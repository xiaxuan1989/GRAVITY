\chapter{黑洞物理}\label{chpt:BH}

黑洞是广义相对论提出之后，首个也是最重要的研究课题。百年来围绕着黑洞的研究，使我们对相对论、引力、量子场论、统计物理都有了崭新的认识。

\section{Birkhoff 定理}

取球坐标 $x^\mu=(\bar t,\bar r,\theta,\phi)$，并以 $A,B=0,1$ 和 $i,j=2,3$ 分别表示 $\bar t,\bar r$ 与 $\theta,\phi$ 两组指标。球对称性是指旋转群 $\mathrm{SO}(3)$ 在每个 $(\bar t,\bar r)$ 不变的 2-球面上等距作用，其三个 Killing 矢量场可取为
\[
    \xi_{(1)}=\p_\phi,\qquad
    \xi_{(2)}=-\cos\phi\,\p_\theta+\cot\theta\sin\phi\,\p_\phi,\qquad
    \xi_{(3)}=\sin\phi\,\p_\theta+\cot\theta\cos\phi\,\p_\phi.
\]
它们分别满足 Killing 方程
\[
    0=(\Lie_{\xi_{(a)}}g)_{\mu\nu}
    =\xi_{(a)}^\lambda\p_\lambda g_{\mu\nu}
    +g_{\lambda\nu}\p_\mu\xi_{(a)}^\lambda
    +g_{\mu\lambda}\p_\nu\xi_{(a)}^\lambda
    =\cd_\mu\xi^{(a)}_\nu+\cd_\nu\xi^{(a)}_\mu.
\]
首先代入 $\xi_{(1)}$ 得 $\p_\phi g_{\mu\nu}=0$。再将 $\xi_{(2)},\xi_{(3)}$ 代入，并分别比较 $\sin\phi,\cos\phi$ 的系数，$AB,Ai,ij$ 分量依次给出
\[
    \p_\theta g_{AB}=0,\qquad
    g_{A2}=g_{A3}=0,\qquad
    \p_\theta g_{22}=0,\qquad
    g_{23}=0,\qquad
    g_{33}=g_{22}\sin^2\theta.
\]
故只有 $g_{00},g_{01},g_{11},g_{22},g_{33}$ 可能非零；前三者及 $g_{22}$ 对 $\theta,\phi$ 的导数均为零。记 $\bar g_{AB}(\bar t,\bar r)=g_{AB}$、$g_{22}=K(\bar t,\bar r)$，则 $g_{33}=K(\bar t,\bar r)\sin^2\theta$。
于是得到\footnote{此形式在数学上属于\textbf{扭曲积}(warped product)：度规可分离为 2 维和 $(n-2)$ 维部分：$g_{ab}(x^0,x^1)\d x^a\d x^b+K(x^0,x^1)\hat g_{ij}(x^2,\cdots,x^{n-1})\d x^i\d x^j$，其中 $ab$ 取 $0,1$；$ij$ 取 $2,\cdots,n-1$。}：
\[\d s^2=\bar g_{00}\d\bar t^2+\bar g_{11}\d\bar r^2+2\bar g_{01}\d\bar t\,\d\bar r+K\d\Omega^2.\]
然而 $\d\Omega^2$ 系数可无关于坐标时。令 $r^2=K(\bar t,\bar r)$，则 $2r\d r=\pdv{K}{\bar t}\d\bar t+\pdv{K}{\bar r}\d\bar r$，代入
\begin{align*}
    \d s^2&=\bar g_{00}\d\bar t^2+\bar g_{11}\left(\frac{2r\d r-\pdv{K}{\bar t}\d\bar t}{\pdv*{K}{\bar r}}\right)^2+2\bar g_{01}\d\bar t\left(\frac{2r\d r-\pdv{K}{\bar t}\d\bar t}{\pdv*{K}{\bar r}}\right)+r^2\d\Omega^2\\
    &=\left(\bar g_{00}+\bar g_{11}\frac{(\pdv*{K}{\bar t})^2}{(\pdv*{K}{\bar r})^2}-2\bar g_{01}\frac{\pdv*{K}{\bar t}}{\pdv*{K}{\bar r}}\right)\d\bar t^2+\frac{4K\bar g_{11}}{(\pdv*{K}{\bar r})^2}\d r^2\\
    &+\left(\frac{4r\bar g_{01}}{\pdv*{K}{\bar r}}-\frac{4r\bar g_{11}\pdv{K}{\bar t}}{(\pdv*{K}{\bar r})^2}\right)\d\bar t\,\d r+r^2\d\Omega^2\\
    &=\tilde g_{00}\d\bar t^2+\tilde g_{11}\d r^2+2\tilde g_{01}\d\bar t\,\d r+r^2\d\Omega^2.
\end{align*}
这还不是遥远观者对钟出的坐标系，因为 $\tilde g_{01}\ne 0$。为此设 $C(\bar t,r)$ 使得 $\d\hat t=C(\tilde g_{00}\d\bar t+\tilde g_{01}\d r)$。这等价于 $\pdv{r}(C\tilde g_{00})=\pdv{\bar t}(C\tilde g_{01})$，或者
\[
    \pdv{\ln C}{\bar t}=\frac{1}{\tilde g_{01}}\left(\tilde g_{00}\pdv{\ln C}{r}+\pdv{\tilde g_{00}}{r}-\pdv{\tilde g_{01}}{\bar t}\right),
\]
上式是关于 $\ln C$ 的简单偏微分方程，只需给定边界函数 $\ln C(0,r)$ 便可积分求解，必存在不唯一的非平凡解。
代入有
\begin{align*}
    \d s^2&=\tilde g_{00}\left(\frac{\d\hat t/C-\tilde g_{01}\d r}{\tilde g_{00}}\right)^2+\tilde g_{11}\d r^2+2\tilde g_{01}\d r\left(\frac{\d\hat t/C-\tilde g_{01}\d r}{\tilde g_{00}}\right)+r^2\d\Omega^2\\
    &=\frac{\d\hat t^2}{C^2\tilde g_{00}}+\left(\tilde g_{11}-\frac{\tilde g_{01}^2}{\tilde g_{00}}\right)\d r^2+r^2\d\Omega^2=-\e^{2\nu}\d\hat t^2+\e^{2\mu}\d r^2+r^2\d\Omega^2,
\end{align*}
其中 $\mu(\hat t,r),\nu(\hat t,r)$ 可以是复函数。

再利用真空方程确定分量。
使用 $\Gamma^\lambda_{\mu\nu}\d x^\mu\d x^\nu$ 或对 $L=\e^{2\nu}(\dv{\hat t}{\tau})^2-\e^{2\mu}(\dv{r}{\tau})^2-r^2(\dv{\Omega}{\tau})^2$ 变分得
\begin{align*}
    & \Gamma_{00}^0=\dot{\nu}, \quad \Gamma_{01}^0=\nu^{\prime}, \quad \Gamma_{11}^0=\e^{2(\mu-\nu)} \dot{\mu}, \\& \Gamma_{00}^1=\e^{2(\nu-\mu)} \nu^{\prime}, \quad \Gamma_{01}^1=\dot{\mu}, \quad \Gamma_{11}^1=\mu^{\prime}, \quad \Gamma_{22}^1=-r \e^{-2 \mu}, \quad \Gamma_{33}^1=-r \e^{-2 \mu} \sin ^2 \theta, \\
    & \Gamma_{12}^2=\Gamma_{13}^3={1}/{r}, \quad \Gamma_{33}^2=-\sin \theta \cos \theta, \quad \Gamma_{23}^3=\cot \theta.
\end{align*}
其中点撇分别代表 $\p_{\hat t},\p_{r}$。
对 Ricci 张量同样可置
\[R_{\mu\nu}\d x^\mu\d x^\nu=\p_\sigma\Gamma^\sigma_{\mu\nu}\d x^\mu\d x^\nu-\d^2\ln\sqrt{-g}+(\ln\sqrt{-g})_{,\kappa}\Gamma^\kappa_{\mu\nu}\d x^\mu\d x^\nu-\Gamma^\kappa_{\sigma\mu}\Gamma^\sigma_{\kappa\nu}\d x^\mu\d x^\nu,\]
其中 $\sqrt{-g}=\e^{\mu+\nu}r^2\sin\theta$。结果为
\begin{align*}
    R_{00} &= -(\ddot\mu+\dot\mu(\dot\mu-\dot\nu))+\e^{2(\nu-\mu)} \left(\nu''+2\nu'/r+\nu'(\nu'-\mu')\right), \\
    R_{11} &= \e^{2(\mu-\nu)} \left(\ddot\mu+\dot\mu(\dot\mu-\dot\nu)\right)-\nu''+2\mu'/r-\nu'(\nu'-\mu'), \\
    R_{22} &= \e^{-2\mu} \left(\e^{2\mu}-1+r(\mu'-\nu')\right), \quad R_{33} = R_{22} \sin ^2 \theta,\quad R_{01} = 2\dot\mu/r.
\end{align*}
施以真空方程：$R_{01}=0$ 给出 $\dot\mu=0$，从而
$\nu''+2\nu'/r+\nu'(\nu'-\mu')=0, 
-\nu''+2\mu'/r-\nu'(\nu'-\mu')=0, 
\e^{2\mu}-1+r(\mu'-\nu')=0$。
前两者相加得 $\mu'+\nu'=0$，代入后者有 $(r\e^{-2\mu(r)})'=1$，得
\[\e^{-2\mu}=1-{2M}/{r},\quad \e^{2\nu}=\e^{f(\hat t)}(1-{2M}/{r}).\]
其中 $M$ 是常数，$f(\hat t)$ 关于 $r$ 为常数。只需取 $\d t=\e^{f(\hat t)}\d\hat t$ 即有 Schwarzschild 解
\[\d s^2=-\left(1-\frac{2M}{r}\right)\d t^2+\left(1-\frac{2M}{r}\right)^{-1}\d r^2+r^2\d\Omega^2.\]
现列举分量如下以供查阅：
\begin{align*}
    \Gamma_{01}^0&=\frac{M}{r^2}\left(1-\frac{2M}{r}\right)^{-1}, \quad \Gamma_{00}^1=\frac{M}{r^2}\left(1-\frac{2M}{r}\right), \\
    \Gamma_{11}^1&=-\Gamma_{01}^0,\quad\Gamma_{22}^1=-r(1-{2M}/{r}),\quad\Gamma_{33}^1=\Gamma_{22}^1\sin ^2 \theta, \\
    \Gamma_{12}^2&=1 / r,\quad\Gamma_{33}^2=-\sin \theta \cos \theta,\quad\Gamma_{13}^3=1 / r,\quad\Gamma_{23}^3=\cot \theta,
\end{align*}
以及
\begin{align*}
    R_{0101}&=-\frac{2 M}{r^3}, \quad R_{0202}=\frac{M}{r}\left(1-\frac{2M}{r}\right), \quad R_{1212}=-\frac{M}{r}\left(1-\frac{2M}{r}\right)^{-1}, \\
    R_{0303}&=R_{0202} \sin ^2 \theta, \quad R_{1313}=R_{1212} \sin ^2 \theta, \quad R_{2323}=2 M r \sin ^2 \theta.
\end{align*}

Schwarzschild 最初除了真空、球对称假设外还引入了静态条件\footnote{1915 年 11 月 18 日，Schwarzschild 趁着假期出席了 Einstein 于普鲁士科学院所作关于场方程的首次演讲，随后于一战德国战场上给出该解——这是场方程第一个引力解析解。他于数月后病逝。}，但上述讨论表明真空球对称解必是 Schwarzschild 解，这也称为\textbf{Birkhoff 定理}。
Schwarzschild 解的静态写法还有很多。\textbf{各向同性坐标系}就是指度规在其中形如 $\d s^2=g_{00}(\rho)\d t^2+K(\rho)\delta_{ij}\d x^i\d x^j$，其中 $\rho=\sqrt{x^2+y^2+z^2}$。只需取 $r=\rho(1+\frac{M}{2\rho})^2$ 有
\eq{
    \d s^2=-\left(\frac{1-M/2\rho}{1+M/2\rho}\right)^2\d t^2+\left(1+\frac{M}{2\rho}\right)^4(\d x^2+\d y^2+\d z^2).
}
注意，$\rho$ 是各向同性径向坐标，并非面积半径 $r$；事件视界 $r=2M$ 对应 $\rho=M/2$，静态外部区域对应 $\rho>M/2$。

考虑球对称带电体。在源外 $r\ne0$ 处，方程为无源 Einstein--Maxwell 方程
\[
    \cd_\mu F^{\mu\nu}=0,\qquad R_{\mu\nu}=8\pi T_{\mu\nu},\qquad
    T_{\mu\nu}=\frac{1}{4\pi}\left(F_{\mu\sigma}F_\nu{}^\sigma-\frac14g_{\mu\nu}F_{\sigma\lambda}F^{\sigma\lambda}\right).
\]
其中第二式利用了 $T^\mu{}_\mu=0$。沿用上节的球对称度规 $\d s^2=-\e^{2\nu(\hat t,r)}\d\hat t^2+\e^{2\mu(\hat t,r)}\d r^2+r^2\d\Omega^2$，
并且不考虑磁单极，则 $F_{01}$ 是电磁场唯一可能非零的独立分量。注意 $\sqrt{-g}=\e^{\mu+\nu}r^2\sin\theta$，Maxwell 方程的 $\nu=0,1$ 分量分别给出
\[
    \p_r\left(r^2\e^{-\mu-\nu}F_{01}\right)=0,\quad
    \p_{\hat t}\left(r^2\e^{-\mu-\nu}F_{01}\right)=0\im
    F_{01}=\frac{Q}{r^2}\e^{\mu+\nu},\quad F_{\sigma\lambda}F^{\sigma\lambda}=-\frac{2Q^2}{r^4},
\]
其中 $Q$ 是常数。代入能动张量得
\[
    8\pi T_{00}=\frac{Q^2}{r^4}\e^{2\nu},\quad
    8\pi T_{11}=-\frac{Q^2}{r^4}\e^{2\mu},\quad
    8\pi T_{22}=\frac{Q^2}{r^2},\quad
    T_{33}=T_{22}\sin^2\theta,\quad T_{01}=0.
\]
再利用上节算得的 Ricci 张量。$R_{01}=0$ 给出 $\dot\mu=0$，从而 $R_{00}=8\pi T_{00},R_{11}=8\pi T_{11},R_{22}=8\pi T_{22}$ 分别化为
\begin{align*}
    \nu''+2\nu'/r+\nu'(\nu'-\mu')&=\frac{Q^2}{r^4}\e^{2\mu},\\
    -\nu''+2\mu'/r-\nu'(\nu'-\mu')&=-\frac{Q^2}{r^4}\e^{2\mu},\\
    \e^{-2\mu}\left(\e^{2\mu}-1+r(\mu'-\nu')\right)&=\frac{Q^2}{r^2}.
\end{align*}
前两者相加仍得 $\mu'+\nu'=0$，故 $\nu(\hat t,r)=-\mu(r)+h(\hat t)$。代入第三式有
\[
    \left(r\e^{-2\mu(r)}\right)'=1-\frac{Q^2}{r^2},\qquad
    \e^{-2\mu}=1-\frac{2M}{r}+\frac{Q^2}{r^2},
\]
其中将另一积分常数写作 $-2M$，代回前两式亦满足。取 $\d t=\e^{h(\hat t)}\d\hat t$ 消去 $h(\hat t)$，在新坐标下 $F_{01}=Q/r^2$。再取 $A_1=0$ 且令 $A_0$ 在无穷远为零，则 $F_{01}=-\p_rA_0$ 给出 $A_0=Q/r$，因而 $Q$ 解释为中心天体的电荷。
于是得\textbf{Reissner-Nordström 解}
\eq{
\tcbhighmath{
    \d s^2=-\left(1-\frac{2M}{r}+\frac{Q^2}{r^2}\right)\d t^2+\left(1-\frac{2M}{r}+\frac{Q^2}{r^2}\right)^{-1}\d r^2+r^2\d\Omega^2,
}
}
以描述球对称带电黑洞。

\section{三大实验验证}

自 Einstein 论文发表以来，对广义相对论的若干实验验证相继提出，其中最为著名的是引力红移、水星进动和星光偏折，它们都与 Schwarzschild 度规有关。

\subsection{引力红移}

回忆一下。设观者、光源各有 4-速 $U^\mu,V^\mu$。光源在发光时测得 $\omega=-k^\mu V_\mu$，观者在接收时测得 $\omega'=-k^\mu U_\mu$。记 $\gamma = - V^\mu U_\mu$，沿 $V^\mu$ 做 3+1 分解 $k^\mu=\omega V^\mu + \bm k^\mu,U^\mu = \gamma V^\mu + \gamma \bm u^\mu$。则
\[\omega'=-\gamma (\omega V_\mu + \bm k_\mu) (V^\mu + \bm u^\mu) = \gamma (\omega - \bm k_\mu \bm u^\mu) = \gamma \omega (1 - u \cos \theta),\]
其中 $u=|\bm u|$，$\theta$ 为（光源系中）$\bm k,\bm u$ 夹角。此即\textbf{相对论 Döppler 效应}，展开级数第一项 $\omega (1 - u \cos \theta)$ 是非相对论 D\"oppler 效应。

置两个稳态观者，一者为光源 $e$，一者为接收端 $r$。二者 4-速共线于 Killing 矢量 $\xi^\mu = \pdv*{x^\mu}{t}$，$t$ 为稳态时间。$k^\mu$ 走测地线，知 $k^\mu\xi_\mu$ 沿光路守恒。
角频率为 $\omega=-k_\mu U^\mu=-k_\mu\xi^\mu/\sqrt{-\xi_\mu \xi^\mu}$，而波长为 $\lambda = 2\pi/\omega$。若 $\{x\}$ 的坐标时就是稳态时间，$\xi^\mu=\delta^\mu_0$，则 $\xi_\mu\xi^\mu=g_{00}$。
或者，注意 $\omega=-\dv*{\theta}{\tau}$，也将发现 $\omega_e/\omega_r=\d\tau|_r/\d\tau|_e=\sqrt{g_{00}|_r/g_{00}|_e}$。故
\eq{
\tcbhighmath{
    z:=\frac{\lambda_r}{\lambda_e}-1=\sqrt{\frac{\xi_\mu\xi^\mu |_r}{\xi_\mu\xi^\mu |_e}}-1 \xRightarrow{\text{Schwarzschild}} \sqrt{\frac{1-2M/r_r}{1-2M/r_e}}-1,
}
}
这称为\textbf{红移量}。$\chi = \sqrt{-\xi_\mu\xi^\mu}$ 称为\textbf{红移因子}。
若 $r_r>r_e$，则 $z>0$，光将红移，否则蓝移。但这并非因为光爬出引力势阱而消耗能量，只是因为引力钟慢效应。被高处静止观测者测得的频率更低。
弱场近似时 $\phi=-M/r$，取 $\Delta\phi=\phi_r-\phi_e$，有 $z=(1+2\phi_r)^{1/2}(1+2\phi_e)^{-1/2}-1\approx\Delta\phi$，考察近地表处相隔 $\Delta h$ 的观者，则 $z=g\Delta h$。这一形式也可用等效原理得到：将引力场中的稳态观者近似为平直时空中的加速观者，在平直时空的惯性系下，设从静止向上加速 $g$ 高为 $\Delta h$ 的电梯，起初底部发射光并测得频率 $\omega_e$，光经 $\Delta t=\Delta h + 1/2 g\Delta t^2\approx \Delta h$ 时间到达顶部，顶部这时的速度为 $u=g\Delta t\approx g\Delta h$，则根据 D\"oppler 效应 $\omega_r/\omega_e = \gamma(1-u)$，考虑到 $u\ll 1$，即 $\gamma\approx 1$，因此 $z = \gamma(1+u) - 1 \approx u = g\Delta h$。

在渐近平直平稳时空中，取无穷远归一化的平稳 Killing 矢量 $\xi^a=(\partial_t)^a$。由沿 $\xi^a$ 静止的光源发出、无穷远静止观者接收的光，若其红移在某曲面处趋于无穷，则该曲面称为\textbf{无限红移面}(infinite-redshift surface)。

光子的守恒能量为 $\mathcal E=-k_a\xi^a$，静止观者测得的频率为 $\omega=\mathcal E/\sqrt{-\xi^2}$，故 $1+z=\omega_{\rm em}/\omega_\infty=1/\sqrt{-\xi^2_{\rm em}}$。因此无限红移面满足 $\xi^a\xi_a=0$；在适应平稳对称性的坐标中即 $g_{tt}=0$。

它表示平稳 Killing 矢量由类时变为类光，静止观者在此达到光速极限，故也称\textbf{静止极限面}(stationary-limit surface)。静态时空中它常与事件视界重合，如 Schwarzschild 时空的 $r=2M$；旋转时空中一般不重合，例如 Kerr 时空的无限红移面是能层的外边界，而事件视界位于其内侧。

19 世纪末，水星近日点进动现象被首次观测。毫无干扰的情况下，Newton 理论当然不能解释进动，因此必须要考虑其它因素，如同当年从天王星轨道扰动发现的“笔尖下的行星”海王星那样。由于地球自转轴的进动，从地球上观测水星的进动大部分是参考系的视觉效果，即岁差，我们直接扣除它。其它因素就是其余行星造成的摄动、太阳实际形状以及广义相对论相应。
主要研究摄动即可。
以太阳为参考，水星所处的外引力场写作（取几何制和单位质量）
\[\bm F(r,t)=\bm F_\odot+\bm F_p=-\frac{M_\odot}{r^3}\bm r-\sum_i\frac{M_i(\bm r-\bm R_i)}{|\bm r-\bm R_i|^3},\]
前一项是太阳的贡献，后面是各行星的作用总和。现在研究后者。设 $\bm r,\bm R_i$ 夹角 $\theta_i$，则 $d_i=|\bm r-\bm R_i|=R_i^2(t)+r^2-2R_i(t)r\cos\theta_i(t)$。
忽略其余行星轨道的离心率之影响，即假定做半径 $\bm R_i$ 的圆周运动。设单个行星角速度为 $\omega_i$，将 $\bm F_p$ 在时域上做 Fourier 展开
\[
\bm F_p=- \sum_i \sum_{n\in\Z} {{a_n}(r){\e^{\i n{\omega_i}t}}}\frac{\bm r-\bm R_i}{|\bm r-\bm R_i|},\quad a_n=\int_0^{2\pi/\omega_i}\frac{\d t}{2\pi}{\frac{\omega_i M_i\e^{-\i n\omega_i t}}{{R_i^2 + r^2 - 2R_ir\cos \theta_i}}}.
\]
只保留最低阶的 $n=0$ 并用$\d\theta_i$ 替换$\omega_i\d t$ 就可使引力势同时间无关，得到近似表达
\eq{\bm F_p(r) = - \sum_i  \int_0^{2\pi }\frac{\d\theta}{2\pi}\frac{M_i}{R_i^2 + r^2 - 2{R_i}r\cos \theta  }\frac{\bm r-\bm R_i}{|\bm r-\bm R_i|},}
引入线密度 $\lambda_i=M_i/(2\pi R_i)$，并沿水星初始方向为极轴，在黄道面上建立极坐标系 $(r,\theta)$，便发现上式表明，其余行星对水星的作用相当于 $M_i$ 均匀分布的一个圆环。这符合直觉，毕竟水星的进动极为缓慢，完整进动一周所需时间相较于其它行星周期很长，因此其它行星的作用的确和圆环类似。但对离水星较近的行星（比如金星），二者周期将在同一数量级，因此不便于近似为圆环。可见圆环模型成立条件是 $R_i\gg r$。幸好，除金星外其它行星都离得较远，将金星视作圆环的影响并不致命。问题现在相当于，在太阳引力基础上，计算附加这样一个微扰后行星的进动。
由圆环对称性，可知作用力与 $\bm r$ 同向，即
\[
\bm F_p= \frac{\bm r}{r}\sum_i  \lambda_iR_i\int_0^{2\pi }\d{\theta}\frac{\cos\alpha}{ d_i^2},
\]
可见其他引力的作用可视作朝外的有心力，我们就可直接讨论分量 $F=F_\odot+F_p$。
我们希望先计算出 $F_p$ 与 $r$ 的关系，虽无初等表达，但由于该式本身就具条件 $R_i\gg r$，故可近似计算。一个聪明的做法并非按 $r/R_i$ 展开，而是重构模型本身。将 $(r,\theta)$ 平移以使原点与水星重合，记新坐标 $(d,\alpha)$，则 $R_i^2=r^2+d_i^2+2rd_i\cos\alpha$，
由于 $\alpha=0$ 时 $d_i=R_i-r$，取解 $d_i=-r\cos\alpha+\sqrt{R_i^2-r^2\sin^2\alpha}$。
由 $R_i\gg r$ 知圆环可近似看作按 $\alpha$ 均匀分布，故 $ R_i\d\theta\approx d_i\d\alpha$，于是
\begin{align}
    F_p&=\sum_i \lambda_i \int_0^{2\pi } \frac{R_i\cos\alpha}{d_i^2 }\d{\alpha}\nonumber\\
    &\approx\sum_i \lambda_i \int_0^{2\pi } \frac{\cos\alpha\d{\alpha}}{-r\cos\alpha+\sqrt{R_i^2-r^2\sin^2\alpha}}\nonumber\\
    &=\sum_i \lambda_i \int_0^{\pi } \left(\frac{\cos\alpha\d{\alpha}}{-r\cos\alpha+\sqrt{R_i^2-r^2\sin^2\alpha}}-\frac{\cos\alpha\d{\alpha}}{r\cos\alpha+\sqrt{R_i^2-r^2\sin^2\alpha}}\right)\nonumber\\
    &=2r \int_0^{\pi }\cos^2\alpha\d{\alpha} \sum_i \frac{\lambda_i}{R_i^2-r^2} = \tcbhighmath{\pi r\sum_i \frac{\lambda_i}{R_i^2-r^2}.}
\end{align}

接下来只以初始位置 $(r_0,0)$ 为极轴方向建立极坐标，以研究水星运动。有心力的形式为
$\ddot r-\dot\theta^2r=F(r), 2\dot r\dot\theta+\ddot\theta r=0$。第二式导致角动量守恒，记 $r^2{\dot \theta}=L$。在力学课学过如何从两式消 $t$ 得 Binet 方程
\eq{
\tcbhighmath{
    \dv[2]{\mu}{\theta}+\mu=-\frac{F}{L^2\mu^2}=J(\mu),\quad \mu=1/r.
}
}
默认加入微扰前是圆轨道模型，则 $\dv*{\mu}{\theta}=0,F_0=-{L^2}/{r_0^3}$，从而 $\mu_0=J(\mu_0)$。
然而圆轨道闭合而不一定稳定。稳定即是说，物体运动受到微扰后依然束缚在圆轨道附近。下面寻找稳定条件。在 $\mu_0$ 附近展开 $J(\mu)$，记 $x=\mu-\mu_0$，只保留至一阶可得波动方程
\[
\dv[2]{x}{\theta}+\left(1-\eval{\dv{J}{x}}_{0}\right)x=0.
\]
如果括号为负，$x$ 将随 $\theta$ 指数增长或衰减，显然不再稳定。因此稳定要求括号为正，则 $x$ 将随 $\theta$ 作周期振荡。代入 $J(\mu)$ 定义得
\[\eval{\dv{J}{x}}_{0}=\eval{\dv{\mu}}_{\mu_0}\left(- \frac{F}{L^2\mu^2} \right)= \frac{2F_0}{L^2\mu_0^3} - \frac{1}{L^2\mu_0^2} \eval{\dv{F}{\mu}}_{\mu_0} = -2 - \frac{r_0}{F_0} F'(r_0).\]
这里撇号表示对 $r$ 求导。则稳定轨道要求有心力的形式满足
\eq{
1-\eval{\dv{J}{x}}_{0}=3+\frac{r_0}{F_0} F'(r_0)>0,\quad r_0F_p'(r_0)=\pi r_0 \sum_{i} \lambda_i\frac{R_i^2+r_0^2}{(R_i^2-r_0^2)^2}.
}
在波动方程中，这个括号的含义是频率平方，注意其中是对 $\theta$ 求导而非 $t$，故应是
\eq{
\frac{2\pi}{\Delta\theta}=\sqrt{3+\frac{r_0}{F_0} F'(r_0)}\im\Delta \theta=2\pi\sqrt{\frac{F_{\odot}(r_0)+F_p(r_0)}{F_\odot(r_0)+3F_p(r_0)+r_0F_p'(r_0)}}.
}
其中代入了\footnote{为更便于计算，还可注意 $F_p(r_0)\approx r_0 F_p^{\prime}(r_0) \ll |F_\odot(r_0)|$，则展至 $F_p(r_0) / F_\odot(r_0)$ 和 $F_p^{\prime}(r_0) / F_\odot(r_0)$ 的一阶。} $F=F_{\odot}+F_p$。$\Delta\theta$ 的含义就是一次微扰振荡周期所扫过的角度，也称近点周期(anomalistic period)，对当前近圆轨道 $r\simeq r_0$ 为
\[
T_r=\frac{\Delta\theta}{\dot\theta}=\frac{\Delta\theta r_0^2}{L}=\Delta\theta\sqrt{-\frac{r_0}{F_0}}.
\]
可见只有 $F_\odot$ 时使 $\Delta\theta=2\pi$，相当于又回到原位而无摄动。一旦错开，一次周期内就出现摄动。摄动率为 $\varpi=(\Delta\theta-2\pi)/T_r$。由于 $\delta = \Delta\theta - 2\pi,|\delta|\ll 1$，因此 $T_r=T_\phi(1+\delta/2\pi)$，这里 $T_\phi = 2\pi/\dot\theta$ 是恒星年(sidereal year)\footnote{也称方位角(azimuthal)周期、未摄动零阶周期。某行星的恒星年是太阳在黄道上的视位置相对遥远恒星变化一周所经历时间，也相当于该行星相对于遥远恒星公转一周所经历的时间。千禧年时地球恒星年的平均长度为 365.25636 天。回归年(tropical year)是太阳的平黄经变化一周所经历时间。由于太阳视运动非均匀，选取不同的起点会得到不同的长度。若选取移动春分点作为基准，太阳从春分点出发运行一周再次回到春分点的时间为春分点年。千禧年时回归年的平均长度为 365.24219 天。恒星年比回归年要长 20 分 24.5 秒，这个差异正是地球造成的岁差。}，从而 $\delta/T_r=\delta/T_\phi+O(\delta^2)$，故在一阶摄动的意义下分母的 $T_r$ 可替换为 $T_\phi$。

还原国际单位制，由于 $GM_\odot\approx 1.3271244\times10^{20}\,\mathrm{m^3\,s^{-2}}$，水星恒星年为 $T_\phi=2\pi\sqrt{{r_0^3}/{GM_\odot}}=87.969\,\text{天}$。
代入数据知 $r_0F_p'(r_0)\approx 1.088 \times 10^{16}\,\mathrm{N}$。
而 $F_p(r_0)\approx 7.587 \times 10^{15}\,\mathrm{N}, F_\odot(r_0)\approx -1.318 \times 10^{22}\,\mathrm{N}$。
$\Delta \theta\approx 2\pi(1+9.884\times 10^{-7})$。于是 $\varpi\approx{\pi(1.977\times 10^{-6})}/{87.969\,\text{天}}\approx 532''/\text{世纪}$，
仍与观测值 $575''$/世纪相差 $43''$/世纪。

再考虑广义相对论效应。为此我们要讨论在球对称度规中的测地线。给定初始条件 $(p,Z^\mu)$。易知，总能找到\footnote{设 $p$ 任意位置。先将极点平移至 $p$，此时 $\theta=0$。由于稳态观者一定不是测地线，所以 $Z^i\ne 0$。若 $Z^i$ 垂直于球面，则一定有 $\dv{\theta}{\tau}=0$。否则可绕极轴旋转（经度平移）使本初子午线与 $Z^i$ 的球面投影相切，从而 $\dv{\theta}{\tau}=0$。再将本初子午线（绕经纬度原点）旋转为赤道使得 $\theta=\pi/2$。}球坐标系使 $p$ 处 $\theta=\pi/2$ 且 $\dv{\theta}{\tau}=0$。$x^\lambda = \theta$ 的测地线方程为 $\ddot\theta+\frac{2}{r}\dot r\dot\theta-\sin\theta\cos\theta\dot\phi^2=0$（这里求导是对任意参数），可见 $\theta\equiv\pi/2$ 满足该方程和初始条件，由解的唯一性知质点运动的 $\theta$ 恒定。

记 $f(r)=1-2M/r$，并以点表示对固有时 $\tau$ 求导。稳态与轴对称 Killing 矢量分别给出单位质量能量、角动量守恒：$E=-\pdv*{x^\mu}{t}Z_\mu=f\dot t$，$L=\pdv*{x^\mu}{\phi}Z_\mu=r^2\dot\phi$。在赤道面上，四速度归一化条件为 $-1=-f\dot t^2+f^{-1}\dot r^2+r^2\dot\phi^2$，代入 $E,L$ 得径向方程 $\dot r^2=E^2-f(1+L^2/r^2)=:E^2-V_{\rm eff}(r)$，其中 $V_{\rm eff}=f(1+L^2/r^2)$ 是有效势。

为研究轨道形状，改用 $\phi$ 参数。令 $u(\phi)=1/r(\phi)$，并以撇表示对 $\phi$ 求导；由 $\dot\phi=L/r^2$ 得 $\dot r=-Lu'$。代回径向方程有
\eq{
L^2(u')^2
=E^2-\Bigl(1-2Mu\Bigr)\Bigl(1+L^2u^2\Bigr)
= (E^2-1)+2Mu-L^2u^2+2ML^2u^3.
}
对 $\phi$ 求导并约去 $2L^2u'$，即得 Binet 方程 $u''+u=M/L^2+3Mu^2$。舍去相对论修正 $3Mu^2$ 就是 Newton 情形，其椭圆解为 $u_0=p^{-1}(1+e\cos\phi)$，其中 $p=L^2/M$ 是半通径，$|e|<1$ 是离心率。

现置 $u=u_0+\delta u$。保留首阶相对论修正，得
\[
\delta u''+\delta u=\frac{3M}{p^2}\left(1+2e\cos\phi+e^2\cos^2\phi\right).
\]
右端的共振项为 $6Me\cos\phi/p^2$，相应特解为 $3Me\phi\sin\phi/p^2$。另一方面，将零阶解的相位换成 $(1-\delta)\phi$，展至 $\delta$ 一阶便产生 $e\delta\phi\sin\phi/p$；比较二者得 $\delta=3M/p=3M^2/L^2$。故径向周期满足 $(1-\delta)\Delta\phi=2\pi$，每绕行一周的近日点进动为 $\Delta\phi_{\rm prec}=\Delta\phi-2\pi\simeq2\pi\delta=6\pi M/p$。

又因 $p=a(1-e^2)$，其中 $a$ 是半长轴，令 $M\mapsto GM/c^2$ 恢复国际单位，便有 $\Delta\phi_{\rm prec}\simeq6\pi GM/[c^2a(1-e^2)]$。代入水星轨道半长轴和太阳质量，得进动约 $43''$/世纪。

\subsection{星光偏折}

星体附近的光线因引力偏离直线，由此形成\textbf{引力透镜}(gravitational lensing)。现计算 Schwarzschild 时空中的弱场偏折角。

仍记 $f(r)=1-2M/r$，取赤道面 $\theta=\pi/2$，并以点表示对仿射参数 $\lambda$ 求导。稳态与轴对称 Killing 矢量给出 $E=f\dot t$、$L=r^2\dot\phi$；零测地约束 $g_{\mu\nu}\dot x^\mu\dot x^\nu=0$ 则给出 $\dot r^2=E^2-fL^2/r^2$。定义碰撞参数 $b=L/E$，再令 $u(\phi)=1/r(\phi)$。由 $\dot\phi=L/r^2$ 仍有 $\dot r=-Lu'$，故 $(u')^2=b^{-2}-u^2+2Mu^3$。对 $\phi$ 求导即得光线的 Binet 方程 $u''+u=3Mu^2$。

在弱场条件 $M/b\ll1$ 下，无引力轨道为 $u_0=b^{-1}\sin\phi$，表示一条在 $\phi=\pi/2$ 处与中心相距 $b$ 的直线。置 $u=u_0+\delta u$ 并保留 $M/b$ 首阶，得
\[
\delta u''+\delta u=3M u_0^2=\frac{3M}{b^2}\sin^2\phi=\frac{3M}{2b^2}\left(1-\cos2\phi\right).
\]
取特解 $\delta u=3M/(2b^2)+M\cos2\phi/(2b^2)$，便有
\[
u(\phi)\simeq \frac{1}{b}\sin\phi+\frac{3M}{2b^2}+\frac{M}{2b^2}\cos2\phi
\]
其中齐次解只改变轨道的相位和 $b$，已由对称边界条件吸收。

远离中心天体时 $u\to0$。将入射与出射的渐近方向分别写作 $\phi=-\alpha$、$\phi=\pi+\alpha$，其中 $\alpha\ll1$。取 $\phi=-\alpha$ 代入上式，得 $0\simeq-\alpha/b+2M/b^2$，即 $\alpha\simeq2M/b$；出射端由对称性给出相同偏折。故总偏折角为 $\Delta\phi_{\rm def}=2\alpha\simeq4M/b$。令 $M\mapsto GM/c^2$ 还原国际单位，即 $\Delta\phi_{\rm def}\simeq4GM/(c^2b)$。

对掠过太阳边缘的星光可取 $b\simeq R_\odot$，因而 $\Delta\phi_{\rm def}\simeq4GM_\odot/(c^2R_\odot)\approx1.75''$。若仅将光视作速率为 $c$ 的 Newton 粒子，则只得 $2GM/(c^2b)$；广义相对论的另一半偏折反映了空间曲率的贡献。


\section{时空延拓}

\subsection{Rindler 系}

相对论限制告诉我们，相对于某外部观者的 3-加速不可能恒定。相对论性匀加速指 4-加速费移，因为 4-加速等于共动系 3-加速，故在共动观者看来，即方向和模长同时恒定的 3-加速，故的确是一种匀加速。
4-加速费移说明 $U^\mu,A^\mu$ 所在平面固定，可退化至 2 维来研究。
取惯性系 $\{T,X\}$，设 4-加速模长恒为 $A$，4-速模长恒为 $-1$ 且指向未来，则
\[-\left(\dv{U^0}{\tau}\right)^2+\left(\dv{U^1}{\tau}\right)^2=A^2,\quad U^0=\sqrt{(U^1)^2+1}.\]
设速度初态 $U^1=0$，则 $U^1=\sinh A\tau,U^0=\cosh A\tau$。设位置初态 $T=0,X=x$，则 $T=\frac 1A\sinh A\tau,X=x+\frac 1A\cosh A\tau-\frac 1A$。
可见轨迹为双曲线。
不妨令 $x=1/A$ 使渐近线穿过原点，即 $X=\frac 1A\cosh A\tau$，或者 $X^2-T^2=x^2$。
由 $U^0=\gamma$ 可知快度为 $\theta=A\tau$。定义\textbf{Rindler 系} $\{t,x\}$ 满足
\eq{\tcbhighmath{T=x\sinh At,\quad X=x\cosh At\im \d s^2=-\d T^2+\d X^2=-(Ax)^2\d t^2+\d x^2.}}
坐标网格即双曲线族和过原点直线族。无交叉项说明 Rindler 系是正交系，故可认为 Rindler 系是按 $A=1/x$ 分布的一系列匀加速观者的共动系。

每条 $x=\text{常数}$ 的世界线都是以 $T=\pm X$ 为渐近线的双曲线。这两条类光直线构成匀加速观者的因果边界，称为\textbf{Rindler 视界}。右楔区 $X>|T|$ 与左楔区 $X<-|T|$ 中的任意两点均类空分离，故其中的观者不能相互通信。

Rindler 度规的行列式 $g=-A^2x^2$ 在 $x=0$ 处为零，但闵氏时空处处平直，并无曲率奇性。事实上，坐标变换的 Jacobi 行列式为 $\pdv{(T,X)}{(t,x)}=Ax$，恰在 $x=0$ 失去可逆性；奇性来自坐标系而非时空。Schwarzschild 坐标在 $r=2M$ 处也有类似问题，因而可先借 Rindler 系说明如何越过坐标边界。

现以类光测地线作为坐标线。令 $\d s^2=0$，得 $\d t=\pm\d x/(Ax)$。相应地定义 Rindler 类光坐标 $u,v$ 及闵氏类光坐标 $U,V$：
\begin{align*}
    v&=t+\frac{1}{A}\ln(Ax),&u&=t-\frac{1}{A}\ln(Ax),\\
    V&=T+X=\frac{1}{A}\e^{Av},&U&=T-X=-\frac{1}{A}\e^{-Au}.
\end{align*}
于是 $\d s^2=-\e^{A(v-u)}\d v\,\d u=-\d V\,\d U$，并有 $T=(V+U)/2$、$X=(V-U)/2$。前一表达仍在 $x\to0$ 时退化，后一表达则在 $U=0$ 或 $V=0$ 处保持光滑。

\end{figure}
如图 \ref{fig:rindlerIM}，等 $x$ 线在闵氏图中是双曲线，等 $t$ 线是过原点的射线。指数关系只给出 $U<0,V>0$，即右 Rindler 楔区；$U=0$ 与 $V=0$ 正是两支视界。由于度规 $-\d V\,\d U$ 不要求这些符号限制，可令 $U,V$ 独立取遍 $\R$，从而把坐标域延拓为整个闵氏时空。若只在原变换中补入 $x<0$，所得仅是左楔区，仍会遗漏未来、过去两个类时楔区。

Rindler 图中，类光测地线趋近视界时有 $|t|\to\infty$，但其闵氏仿射参数 $U$ 或 $V$ 有限，故原坐标域内的测地线并不完备；闵氏延拓使它们自然穿过视界继续延伸。整个过程可概括为 $(t,x)\to(u,v)\to(U,V)\to(T,X)$。下文对 Schwarzschild 视界的处理遵循同一思路：先以类光线建立坐标，再消去视界处发散的坐标尺度。

\subsection{Eddington 系}

仍记 $f(r)=1-2M/r$。径向光线满足 $\d\Omega=0=\d s^2$，故 $\d t/\d r=\pm f^{-1}$。定义\textbf{乌龟坐标}(tortoise coordinate) $r_*$ 使 $\d r_*=\d r/f$，并选取积分常数使 $r\to0$ 时 $r_*\to0$，即
\eq{
    r_*=r+2M\ln\abs{\frac{r}{2M}-1},
}
内、外向光线因而分别满足 $t=-r_*+C$、$t=r_*+C$。当 $r\to2M$ 时 $r_*\to-\infty$，有限的径向区间被拉伸为无限坐标距离，这正是“乌龟”之名的含义，
类比 Zeno 悖论\footnote{古希腊哲学家 Zeno 描述了类似效应：运动员 Achilles 欲同一只乌龟赛跑，且乌龟起点先于他若干路程 $L$。按通常的时间 $t$ 来说，他的速度 $v_1$ 快于乌龟的 $v_2$，明显在有限时间 $t=L/(v_1-v_2)$ 内超过乌龟。但设想这样计时：他欲超过乌龟，必须先追至乌龟的起点，但一旦如此，乌龟又能前进若干步，这样便有新的起点等着他，如此这般有无穷多这样的起点，则他“永远”也赶不上乌龟。这种计时方法称为 Achilles 时或 Zeno 时，记作 $t'$。令 $t',t$ 皆起于零。当 $t=L/v_1$ 时，Achilles 到达乌龟在 $t'=0$ 时的起点， $t'$ 加一，以此类推有
\[t=\sum_{i=0}^{t'-1} \frac{L}{v_1}\left(\frac{v_2}{v_1}\right)^{i}=\frac{L}{v_1-v_2}\bigg(1-\bigg(\frac{v_2}{v_1}\bigg)^{t'}\bigg)\im t'=\frac{1}{\ln(v_2/v_1)}\ln\left(1-\frac{v_2-v_1}{L}t\right),\]
其中将 $t'\in\N$ 自然延拓至 $t'\in\R_+$。可见 $t'$ 在 $t=L/(v_2-v_1)$ 发散，因此按 $t'$ 衡量他永不超过乌龟。}。
若需反解，取满足 $y\e^y=x$、$y\geqslant-1$ 的 Lambert 函数 $\operatorname W$，有 $r=2M[1+\operatorname W(\pm\e^{r_*/2M-1})]$，正、负号分别对应视界外、内。

在 $(t,r_*)$ 中，度规虽可写成 $\d s^2=f(-\d t^2+\d r_*^2)+r^2\d\Omega^2$，但其行列式为 $g=-f^2r^4\sin^2\theta$，仍在视界处退化；$r_*$ 只是把视界移至坐标无穷远，并未完成延拓。为穿过视界，须保留有限的 $r$ 并同时改变时间坐标。

先取\textbf{超前类光坐标} $v=t+r_*$，则 $\d t=\d v-\d r/f$，代回得
\eq{
    \d s^2=-f\,\d v^2+2\d v\,\d r+r^2\d\Omega^2.
}
这称为\textbf{超前 Eddington--Finkelstein 系}。其度规分量在 $r=2M$ 处均有限，且 $g=-r^4\sin^2\theta$ 不退化，故可由外部自然延拓至 $0<r<2M$；$r=0$ 的曲率奇性则仍然存在。径向类光线中，内向族满足 $v=\text{常数}$，可直接穿过未来视界；外向族满足 $\d r/\d v=f/2$，在视界处与之相切。交叉项 $g_{vr}=1>0$ 因而对应黑洞的未来演化方向。

同理，取\textbf{推迟类光坐标} $u=t-r_*$ 得
\[
    \d s^2=-f\,\d u^2-2\d u\,\d r+r^2\d\Omega^2.
\]
此即推迟 Eddington--Finkelstein 系：外向光线满足 $u=\text{常数}$，坐标正则于过去视界，描述超前系的时间反演。可见两套坐标各自越过一支视界，而后文的 Kruskal 系将同时正则化两支视界。

\subsection{Novikov 坐标*}

沿测地线汇建立固有系也可自然地消除奇性。
考虑测地线汇从 $t = 0 = \tau$ 释放，每一线用 $r_m$ 标记，新系为 $(\tau,r_m)$。由 $Z^\mu Z_\mu = -1$ 可知
\[\d\tau=-Z_0(r_m)\d t-Z_1(r,r_m)\d r=\sqrt{1-\frac{2M}{r_m}}\d t+\frac{\sqrt{2M/{r}-2M/{r_m}}}{1-{2M}/{r}}\d r.\]
$r = r_m\cos^2\frac{\eta}{2},\tau = -(\eta+\sin\eta){r_m^{3/2}}/{\sqrt{8M}}$ 确定了变换 $r(\tau,r_m)$。
记变换 $\d r = \pdv{r}{\tau}\d\tau + \pdv{r}{r_m}\d r_m$，由测地线 $\tau$ 轴 $\d\r_m = 0$ 可知 $\pdv{r}{\tau} = Z^1(r,r_m)$，从而 $-Z^1(r,r_m)\d\tau+\d r=\cos^2\frac{\eta}{2}\d r_m$。依次做 $t\to\tau,r\to r_m$ 变换有
\[\d s^2 = -\d\tau^2+\left(\frac{-Z^1(r,r_m)\d\tau+\d r}{Z_0(r_m)}\right)^2+r^2\d\Omega^2= -\d\tau^2+\frac{\cos^4({\eta}/{2})\d r_m^2}{1-2M/r_m}+r^2\d\Omega^2.\]
其中加上了没动过的空间部分。$\eta(\tau,r_m)$ 表明这是非稳态系，直观上落体者当然感受到变化的引力场。
$r_m$ 称\textbf{Novikov 径向坐标}。可以看到的确 $\tilde g_{00}=-1,\tilde g_{01}=0$。
物理意义上，前者直接因为 $Z^\mu Z_\mu = -1$；后者是因为测地线汇从 $\tau = 0$ 静止释放且 $Z^\mu Z_\mu = -1$，由 \ref{sec:tidal} 节知一定保持正交。

实际上也可先从 $\tilde g_{01}=0$ 出发，利用 $\pdv{\tilde x^\mu}{x^\nu}\tilde Z_\mu=Z_\nu$ 得 $\pdv{\tau}{t}=-Z_0,\pdv{\tau}{r}=-Z_1$；
记 $\d R=A\d t+B\d r$，代入 $\tilde Z^1=\pdv{R}{x^\nu}Z^\nu$ 可知 $B=-AZ^0/Z^1$，则 $\d R=A(\d t-\frac{Z^0}{Z^1}\d r)=A(\frac{1}{-Z_0}(\d\tau+Z_1\d r)-\frac{Z^0}{Z^1}\d r)=\frac{A}{Z_0Z^1}(-Z^1\d\tau+\d r)$。
可见 $R$ 可确定到只差缩放函数。
一个更方便的假设是所有测地线从无穷远释放并令 $A=1$（$R$ 沿 $t$ 均匀），因而
\begin{align}
    \d\tau &= \d t+\frac{\sqrt{2M/r}}{1-{2M}/{r}}\d r\im \tau = t+2 \sqrt{2Mr}+2M\ln\abs{\frac{\sqrt{r}-\sqrt{2M}}{\sqrt{r}+\sqrt{2M}}},\\
    \d R &= \d\tau+\sqrt{\frac{r}{2M}}\d r = \d t+\frac{\sqrt{r/2M}}{1-{2M}/{r}}\d r\im R=\tau+\frac 23(2M)^{-1/2}r^{3/2},
\end{align}
则
\begin{align}
    \d s^2 &= -\d\tau^2+\left(\sqrt{\frac{2M}{r}}\d\tau+\d r\right)^2+r^2\d\Omega^2\\
    &= -\d\tau^2+\frac{2M}{r}\d R^2+r^2\d\Omega^2,\quad r=(2M)^{1/3}\left(\frac 32(R-\tau)\right)^{2/3}.
\end{align}
前者为\textbf{Gullstrand–Painlevé 系}，$r=2M$ 时分量收敛且 $g\ne 0$ 表明已消除奇性，但还不是测地固有系； 
后者为\textbf{Lemaître 系}。可以看到 $\tau=R$ 对应 $r=0$。

\subsection{Kruskal 系}

超前、推迟 Eddington--Finkelstein 系各自只正则化一支视界，并以时空交叉项换取消除坐标奇性。若要同时越过两支视界且保持径向度规对角，可仿照 Rindler 延拓，对两个类光坐标一齐作指数变换。

先在外部区域 $r>2M$ 考虑 $v=t+r_*$、$u=t-r_*$。Schwarzschild 度规的径向部分为 $\d s_{(2)}^2=-f\,\d v\,\d u$。引入待定常数 $\beta>0$，令
\[
    V=\e^{\beta v}>0,\qquad U=-\e^{-\beta u}<0,
\]
则 $\d s_{(2)}^2=-\beta^{-2}f\e^{\beta(u-v)}\d V\,\d U$。因 $u-v=-2r_*$，有
\[
    \e^{\beta(u-v)}
    =\e^{-2\beta r}\left(\frac{r}{2M}-1\right)^{-4\beta M}.
\]
要消去 $f=(r-2M)/r$ 在视界处的一阶零点，必须取 $\beta=1/(4M)$。于是
\[
    \d s_{(2)}^2=-\frac{32M^3}{r}\e^{-r/2M}\d V\,\d U.
\]
再定义 $T=(V+U)/2$、$X=(V-U)/2$，则 $-\d V\,\d U=-\d T^2+\d X^2$。还原角向部分便得\textbf{Kruskal 度规}
\eq{\label{eq:kruskal}
    \d s^2=\frac{32M^3}{r}\e^{-r/2M}(-\d T^2+\d X^2)+r^2\d\Omega^2.
}
其系数在 $r=2M$ 处有限且非零，故两支视界都不再是坐标奇点。

在外部区域，变换还可写成
\[
    T=\e^{r/4M}\sqrt{\frac{r}{2M}-1}\sinh\frac{t}{4M},\qquad
    X=\e^{r/4M}\sqrt{\frac{r}{2M}-1}\cosh\frac{t}{4M}.
\]
因此等 $r$ 线是双曲线，等 $t$ 线是过原点的射线。由坐标定义又有
\eq{\label{eq:kruskal-r}
    X^2-T^2=-UV
    =\left(\frac{r}{2M}-1\right)\e^{r/2M}.
}
故 $r$ 是 $T,X$ 的函数；以 Lambert 函数反解即
\eq{\label{eq:r=T,X}
    r=2M\left[1+\operatorname W\left(\frac{X^2-T^2}{\e}\right)\right].
}
径向二维度规只是二维 Minkowski 度规乘上正函数，因而在 $T$--$X$ 图中类光线斜率仍为 $\pm1$。这里仅有径向二维部分共形平直，并不意味着完整四维 Schwarzschild 时空共形平直。
\begin{figure}[h!]
    \centering
    \includegraphics[width=.6\textwidth]{fig/chpt02/kruskal.pdf}
    \caption{Kruskal 延拓及其四个区域}
    \label{fig:kruskal}
\end{figure}
式 \eqref{eq:kruskal-r} 在视界内仍有光滑实解，因此可解除 $U<0,V>0$ 的限制。由 $r>0$ 得 Kruskal 坐标域 $X^2-T^2>-1$，它被 $U=0$ 与 $V=0$ 两条视界分成图 \ref{fig:kruskal} 所示四区：右侧 \textbf{A 区}满足 $U<0,V>0$，是原 Schwarzschild 外部；上方 \textbf{B 区}满足 $U>0,V>0$，是黑洞内部；左侧 \textbf{A$'$ 区}满足 $U>0,V<0$，是第二个渐近平直外部；下方 \textbf{W 区}满足 $U<0,V<0$，是白洞内部。

超前 Eddington--Finkelstein 系保持 $V>0$ 而允许 $U$ 变号，故覆盖 $\mathrm A\cup\mathcal H_V^+\cup\mathrm B$；其中可统一写成 $V=\e^{(t+r_*)/4M}$、$U=-\sgn(r-2M)\e^{(r_*-t)/4M}$。相反，推迟系保持 $U<0$ 而允许 $V$ 变号，覆盖 $\mathrm A\cup\mathcal H_U^-\cup\mathrm W$。Kruskal 系则令 $U,V$ 独立变化，从而同时包含四个区域及四支视界。离开视界后，Schwarzschild 时间可由 $t=2M\ln|V/U|$ 恢复。

B 区中所有指向未来的类时、类光曲线最终都到达 $r=0$；W 区是其时间反演，因果曲线只能从其中穿出。A 与 A$'$ 都渐近平直，却彼此无因果联系。在 $T=0$ 的空间切片上，两外部区域由 Einstein--Rosen 桥连接，其最小面积球面位于 $U=V=0$、$r=2M$，称为\textbf{喉}(throat)或\textbf{分叉球面}(bifurcation sphere)；该桥不可穿越。

边界 $r=0$ 对应 $T^2-X^2=1$。此处曲率不变量 $R_{\mu\nu\rho\sigma}R^{\mu\nu\rho\sigma}=48M^2/r^6$ 发散，不能再以光滑坐标延拓。除此之外，区域 $X^2-T^2>-1$ 已无有限坐标边界，故所得时空称为 Schwarzschild 解的\textbf{最大解析延拓}，简称\textbf{Kruskal 延拓}。

上述白洞、第二外部区域及 Einstein--Rosen 桥属于永恒真空 Schwarzschild 解的整体结构。由恒星坍缩形成的实际黑洞一般只包含一个外部区域与未来黑洞区域，并不产生完整的白洞和第二外部区域。

\subsection{Penrose 图}\label{sec:penrose-diagram}

Kruskal 坐标消除了视界的坐标奇性，却仍让无穷远位于无限坐标处。若只关心因果结构，可再作共形变换，将整个时空压缩到有限图形中。根据 \S\ref{sec:cosmic-censorship}，在物理时空 $(M,g)$ 上取正函数 $\Omega$，令 $\tilde g=\Omega^2g$；由于 $\tilde g_{\mu\nu}k^\mu k^\nu=\Omega^2g_{\mu\nu}k^\mu k^\nu$，类光方向及光锥均不改变。若 $\Omega$ 在扩大的流形边界上趋于零，而 $\tilde g$ 可光滑延伸至该边界，所得构造就是共形紧化，其有限图示称为\textbf{Penrose 图}。

球对称时空的 Penrose 图只画 $(t,r)$ 所张成的径向二维商空间，每个内部点实际代表一个对称 2-球。对 Kruskal 类光坐标作
\[
    \hat U=\arctan U,\qquad \hat V=\arctan V,\qquad
    \mathcal T=\frac{\hat V+\hat U}{2},\qquad
    \mathcal X=\frac{\hat V-\hat U}{2},
\]
则 $\hat U,\hat V\in(-\pi/2,\pi/2)$，原来无界的 $U,V$ 被压入有限区间。径向度规变为某个正函数乘以 $-\d\hat V\,\d\hat U=-\d\mathcal T^2+\d\mathcal X^2$，故略去该共形因子后，径向类光线仍处处与坐标轴成 $45^\circ$。
由式 \eqref{eq:kruskal-r}，物理区域满足 $UV<1$。在 B 区的奇性 $UV=1$ 上，$\arctan U+\arctan V=\pi/2$，故 $\mathcal T=\pi/4$；W 区同理给出 $\mathcal T=-\pi/4$。于是两支 $r=0$ 奇性成为水平类空边界，而 $U=0$、$V=0$ 两支视界仍为对角线。无穷远则被压到外侧四条类光边和六个端点。

    \centering
    \includegraphics[width=.75\textwidth]{fig/chpt02/penrose.pdf}
    \caption{最大延拓 Schwarzschild 时空的 Penrose 图}
    \label{fig:sch-penrose}
\end{figure}

图 \ref{fig:sch-penrose} 中，$\mathscr I^\pm$ 就是 \S\ref{sec:cosmic-censorship} 提过的类光无穷远，是出射、入射类光线的理想端点；$i^\pm$ 是类时世界线的未来或过去端点，称为\textbf{类时无穷远}；$i^0$ 是类空曲线趋向的\textbf{类空无穷远}。A 与 A$'$ 各有一套这样的渐近边界，反映两个渐近平直外部区域。

这些无穷远点是共形延拓所附加的理想边界，并不属于物理时空。$r=0$ 也不属于时空，但原因不同：它是曲率发散的真实奇性，不能作为光滑共形边界补入。视界 $\mathcal H^\pm$ 则是时空内部的正则类光超曲面，只是在图上恰好构成不同区域的分界。

Penrose 图使事件视界的整体性质尤为清楚：相对于某一外部区域的 $\mathscr I^+$，能把因果信号送至该类光无穷远的事件构成其因果过去；这一集合的边界就是未来事件视界 $\mathcal H^+$。因此视界不能只凭某处的曲率或局域测量确定，而依赖整个时空的未来。B 区中未来不可延的类时、类光测地线终止于上方奇性，不能到达任一 $\mathscr I^+$；W 区则只能向外发出因果线。

共形紧化保留的是光锥、因果先后和边界类型，并不保留距离、固有时或曲率大小。图上有限的线段完全可能代表无限仿射距离。对于由坍缩形成的黑洞，W 区与 A$'$ 区不再出现；其 Penrose 图将在未来视界左侧以星体内部取代，这也说明永恒 Schwarzschild 图描述的是理想化的整体真空解。

\subsection{嵌入图与虫洞}

Penrose 图强调因果结构，另一种互补的图示则强调某个空间切片的内禀几何。固定 Schwarzschild 时间，并在球对称空间中取赤道面 $\theta=\pi/2$，其诱导度规为
\[
    \d \ell^2=\frac{\d r^2}{1-2M/r}+r^2\d\varphi^2,
    \qquad r\geqslant 2M.
\]
为了把这张弯曲的二维曲面画出来，设它等距嵌入 Euclidean 三维空间。以 $(r,\varphi,z)$ 为柱坐标，并令曲面由 $z=z(r)$ 给出，则曲面上的线元是
\[
    \d \ell_E^2=\left[1+\left(\dv{z}{r}\right)^2\right]\d r^2+r^2\d\varphi^2.
\]
逐项比较两式可得
\[
    \dv{z}{r}=\pm\sqrt{\frac{2M}{r-2M}},\qquad
    z=\pm2\sqrt{2M(r-2M)},
\]
其中已选择 $z=0$ 对应 $r=2M$。等价地，$r=2M+z^2/(8M)$；这张旋转曲面称为\textbf{Flamm 抛物面}(Flamm paraboloid)。远处 $\dv*{z}{r}\to0$，曲面渐近平坦；在 $r=2M$ 处斜率发散，形成最窄的圆周。

\[
    l(r)=\pm\int_{2M}^{r}\frac{\d\bar r}{\sqrt{1-2M/\bar r}}
    =\pm\left[\sqrt{r(r-2M)}+2M\ln\frac{\sqrt r+\sqrt{r-2M}}{\sqrt{2M}}\right].
\]
取正、负号分别得到 A 与 A$'$ 两个外部区域。以 $l$ 为坐标时，完整等时空间的度规可写成 $\d\ell^2=\d l^2+r^2(l)\d\Omega^2$；$r(l)=r(-l)$ 在 $l=0$ 取得光滑最小值 $2M$，而 $l\to\pm\infty$ 给出两个渐近平直端。因此该空间切片的拓扑为 $\mathbb R\times S^2$，连接两端的部分称为\textbf{Einstein--Rosen 桥}，也是最早的\textbf{虫洞}(wormhole)实例。图 \ref{fig:flamm-paraboloid} 只嵌入了其中的赤道面；黑色圆周在完整三维切片中实际是面积为 $16\pi M^2$ 的喉球面。

空间上连通并不意味着可以在时空中穿行。这里的喉球面正是未来、过去视界相交的分叉球面；由图 \ref{fig:sch-penrose} 可见，从 A 区出发的未来因果曲线若穿过未来视界，便进入 B 区并终止于 $r=0$，不可能到达 A$'$。故 Schwarzschild 的 Einstein--Rosen 桥是\textbf{不可穿越虫洞}(non-traversable wormhole)。嵌入图没有时间方向和光锥，单凭其漏斗形状无法判断这一因果性质。

这一点也可由一般静态球对称虫洞的写法看出：
\[
    \d s^2=-\e^{2\Phi(r)}\d t^2
    +\frac{\d r^2}{1-b(r)/r}+r^2\d\Omega^2.
\]
$b(r)$ 称为\textbf{形状函数}(shape function)，喉半径 $r_0$ 满足 $b(r_0)=r_0$；赤道面的嵌入方程为 $\dv*{z}{r}=\pm\sqrt{b/(r-b)}$。曲面在喉部向外张开还要求 $b'(r_0)<1$。若要可穿越，红移函数 $\Phi$ 必须处处有限，使 $\e^{2\Phi}$ 不出现视界。Schwarzschild 情形虽有 $b=2M$，却有 $\e^{2\Phi}=1-2M/r$，恰在喉部为零，因而不满足可穿越条件。

在经典 Einstein 方程下，若 $\Phi$ 在喉部正则，场方程还给出
\[
    8\pi(\rho+p_r)\big|_{r_0}=\frac{b'(r_0)-1}{r_0^2}<0,
\]
其中 $\rho$ 与 $p_r$ 是静止观者测得的能量密度和径向压强。于是径向类光方向上的零能量条件在喉部遭到破坏；维持静态可穿越虫洞通常需要所谓的\textbf{奇异物质}(exotic matter)。这项结论涉及完整时空度规，而不仅是某张嵌入图。

最后，嵌入图依赖空间切片的选择，并非时空本身唯一不变的画像。例如 Gullstrand--Painlevé 坐标的等时空间是平直的，便没有上述漏斗形状；但事件视界与因果关系并未因此消失。因此嵌入图适合展示指定切片上的距离和喉部几何，Penrose 图才负责展示虫洞能否穿越。


\section{引力坍缩}\label{sec:collapse}

静态 Schwarzschild 解说明真空黑洞可以存在，却没有说明它怎样从普通物质形成。恒星失去支撑后持续收缩，密度与时空曲率随之增大，这一动力学过程称为\textbf{引力坍缩}(gravitational collapse)。坍缩并不必然产生黑洞：质量较小的星体可以形成白矮星或中子星，外层物质也可能在爆发中被抛出。只有当剩余物质不能建立新的稳定平衡，并收缩到形成陷获区域时，黑洞才会出现。

\subsection{从静力平衡到持续坍缩}

球对称理想流体的静态平衡由 Tolman--Oppenheimer--Volkoff 方程控制。在几何单位制中，令 $m(r)$ 为半径 $r$ 内的质量、$\rho$ 为能量密度、$p$ 为压强，则
\[
    \dv{m}{r}=4\pi r^2\rho,\qquad
    \dv{p}{r}=-\frac{(\rho+p)(m+4\pi r^3p)}{r(r-2m)}.
\]
给定状态方程 $p=p(\rho)$ 与中心密度后，可以向外积分至 $p=0$ 的星体表面。沿一族平衡解增大中心密度时，稳定性通常会在某个转折点改变；若总质量超过相应物态所能支撑的范围，压强梯度便不足以抵抗引力，星体开始动力学坍缩。这里的临界质量依赖状态方程、温度、自转和物质损失，不能只由 Schwarzschild 半径一个量决定。

球对称星体的表面半径记作 $R(\tau)$。表面之外是真空，故由 Birkhoff 定理，外部度规在整个球对称运动过程中仍是质量参数为 $M$ 的 Schwarzschild 度规；时间依赖集中在星体内部以及表面在该外部几何中的运动。自由下落的表面可在有限固有时内越过 $R=2M$。用外部 Schwarzschild 时间描述时，表面却只在 $t\to\infty$ 时趋近 $2M$，同时其发出的光越来越红、到达间隔越来越长。这只是 Schwarzschild 坐标在视界上的退化，并不表示坍缩在物理上停止。

\subsection{Oppenheimer--Snyder 模型}

最简单的精确模型把星体理想化为均匀、无压的尘埃球。尘埃粒子沿类时测地线运动，内部是闭的 FLRW 度规的一部分：
\[
    \d s^2=-\d\tau^2+a^2(\tau)
    \left(\d\chi^2+\sin^2\chi\,\d\Omega^2\right),
    \qquad 0\leqslant\chi\leqslant\chi_b.
\]
$\chi=\chi_b$ 是星体表面，面积半径为 $R(\tau)=a(\tau)\sin\chi_b$。尘埃密度满足 $\rho a^3=\text{常数}$，Friedmann 方程为
\[
    \dot a^2=\frac{8\pi}{3}\rho a^2-1.
\]
若尘埃球在最大半径 $a_{\max}$ 处从静止开始坍缩，可引入参数 $0\leqslant\eta\leqslant\pi$，写成
\[
    a(\eta)=\frac{a_{\max}}{2}(1+\cos\eta),\qquad
    \tau(\eta)=\frac{a_{\max}}{2}(\eta+\sin\eta).
\]
因此 $a$ 在固有时 $\tau_s=\pi a_{\max}/2$ 变为零。这个模型中所有尘埃壳层同时到达 $R=0$，密度和曲率发散，形成类空奇性。

把内部解与外部 Schwarzschild 解在星体表面匹配，得到守恒的总质量
\[
    M=\frac{4\pi}{3}\rho R^3
     =\frac{a_{\max}}{2}\sin^3\chi_b.
\]
表面越过其 Schwarzschild 半径的条件 $R=2M$ 给出
\[
    \eta_H=\pi-2\chi_b\qquad(0<\chi_b<\pi/2),
\]
显然发生在 $\eta=\pi$ 的奇性之前。Oppenheimer--Snyder 模型忽略了压强、密度梯度、自转和辐射，却清楚展示了一个关键事实：正则初始数据可以在有限固有时内演化出视界和奇性，外部则始终保持 Schwarzschild 形式。

\subsection{视界的形成}

视界面积的演化由类光测地线汇控制。

类光聚焦引理是面积定理与奇性定理的共同局域基础。

表观视界是准局域概念，可以由某一时刻的几何求出，但其位置依赖空间切片。事件视界则由整个时空定义：
\[
    \mathcal B=M\setminus J^-(\mathscr I^+),\qquad
    \mathcal H^+=\partial J^-(\mathscr I^+),
\]
其中 $\mathcal B$ 是黑洞区。要判断一束光最终能否到达 $\mathscr I^+$，原则上必须知道时空的全部未来。因此事件视界可以穿过局部曲率有限、尚未出现陷获面的区域。在球对称坍缩图中，它从正则中心发出，穿过星体表面后成为外部的 $r=2M$ 类光超曲面。

\section{KN 黑洞}\label{sec:KN}

前述 Schwarzschild 与 Reissner--Nordstr\"om 解都是非旋转的，后者只是在前者的基础上加入电荷，而实际天体还可具有角动量。在四维、渐近平直的 Einstein--Maxwell 理论中，描述稳态旋转带电黑洞的精确解是 \textbf{Kerr--Newman 解}，简称 KN 解。本节继续采用 $G=c=1$，不考虑磁单极；其三个参量将分别解释为质量 $M$、角动量 $J$ 和电荷 $Q$。

\subsection{轴对称度规}

稳态轴对称时空具有两个对易 Killing 矢量
\[
    \xi=\p_t,\qquad \psi=\p_\phi,
    \qquad [\xi,\psi]=0,
\]
其中 $\xi$ 在无穷远取单位时间平移，$\psi$ 的轨道以 $2\pi$ 为周期，并在旋转轴上退化为零。对于满足正交传递性（又称圆性条件）的 Einstein--Maxwell 解，可选取适配坐标 $(t,\phi,x^1,x^2)$，使度规写成
\[
    \d s^2=g_{tt}\d t^2+2g_{t\phi}\d t\,\d\phi+g_{\phi\phi}\d\phi^2
    +h_{AB}\d x^A\d x^B,
\]
所有系数只依赖 $x^1,x^2$。进一步可在二维轨道空间中选取正交坐标，将 $h_{AB}$ 对角化。这里的 $g_{t\phi}$ 是旋转的关键：它使局域惯性系相对于无穷远发生拖曳。一般含任意物质的轴对称度规未必满足圆性条件，故不能仅凭对称性就删去所有 $t,x^A$ 与 $\phi,x^A$ 交叉项。

保持 $x^A$ 不变、以角速度 $\Omega=\d\phi/\d t$ 运动的稳态观者具有 4-速度 $u=u^t(\xi+\Omega\psi)$。其类时条件为
\[
    g_{tt}+2\Omega g_{t\phi}+\Omega^2g_{\phi\phi}<0,
\]
这将在下文同时刻画能层中的拖曳和视界上的共同角速度。

\subsection{Kerr--Newman 度规}

取 Boyer--Lindquist 坐标 $(t,r,\theta,\phi)$，定义
\begin{equation}\label{eq:kn-functions}
    \Sigma=r^2+a^2\cos^2\theta,\qquad
    \Delta=r^2-2Mr+a^2+Q^2,
\end{equation}
则 Kerr--Newman 度规为（证明从略）
\begin{equation}\label{eq:kn-metric}
\begin{split}
    \d s^2={}&-\frac{\Delta}{\Sigma}
       (\d t-a\sin^2\theta\,\d\phi)^2
       +\frac{\sin^2\theta}{\Sigma}
       \bigl[(r^2+a^2)\d\phi-a\d t\bigr]^2\\
       &+\frac{\Sigma}{\Delta}\d r^2+\Sigma\d\theta^2 .
\end{split}
\end{equation}
与前文 $A_t\sim Q/r$ 的约定相容，电磁势可取为
\begin{equation}\label{eq:kn-potential}
    A=\frac{Qr}{\Sigma}
       (\d t-a\sin^2\theta\,\d\phi),
    \qquad F=\d A .
\end{equation}
电势的整体正负可吸收到 $Q$ 的定义中，度规则只依赖 $Q^2$。展开 \eqref{eq:kn-metric} 的 $t,\phi$ 部分并记
\[
    \mathcal A=(r^2+a^2)^2-a^2\Delta\sin^2\theta,
\]
有
\begin{equation}\label{eq:kn-components}
\begin{split}
    g_{tt}&=-\left(1-\frac{2Mr-Q^2}{\Sigma}\right),\\
    g_{t\phi}&=-\frac{a(2Mr-Q^2)\sin^2\theta}{\Sigma},\\
    g_{\phi\phi}&=\frac{\mathcal A\sin^2\theta}{\Sigma},\qquad
    \det(g_{\mu\nu})=-\Sigma^2\sin^2\theta .
\end{split}
\end{equation}

在 $r\to\infty$ 时，$g_{tt}=-1+2M/r+O(r^{-2})$、$g_{t\phi}=-2Ma\sin^2\theta/r+O(r^{-2})$ 且 $A_t=Q/r+O(r^{-3})$，故 ADM 质量、角动量与电荷分别为
\[
    M,\qquad J=Ma,\qquad Q.
\]
$a$ 因而是单位质量角动量。置 $Q=0$ 得 Kerr 解，置 $a=0$ 得 Reissner--Nordstr\"om 解，再置 $a=Q=0$ 便退化为 Schwarzschild 解。Boyer--Lindquist 坐标中的 $r$ 不是面积半径；$r=\text{常数}$ 的二维曲面一般也不是圆球。

由 $g^{rr}=\Delta/\Sigma$，$\Delta$ 的零点是候选类光超曲面：
\begin{equation}\label{eq:kn-horizons}
    r_\pm=M\pm\sqrt{M^2-a^2-Q^2}.
\end{equation}
当 $M^2>a^2+Q^2$ 时有两支不同的根，$r_+$ 是外事件视界，$r_-$ 是内 Cauchy 视界；这称为\textbf{亚极端}情形。当 $M^2=a^2+Q^2$ 时两根合并为 $r=M$，得到\textbf{极端}黑洞，其视界退化且外部空间切片具有无限长的喉。当 $M^2<a^2+Q^2$ 时 $\Delta$ 无实根，环奇性暴露给无穷远，所得超极端解不是黑洞。

$\Sigma/\Delta$ 在 $r=r_\pm$ 发散只是 Boyer--Lindquist 坐标奇性。引入内落坐标
\begin{equation}\label{eq:kn-ingoing}
    \d v=\d t+\frac{r^2+a^2}{\Delta}\d r,
    \qquad
    \d\tilde\phi=\d\phi+\frac{a}{\Delta}\d r,
\end{equation}
度规便可光滑穿过未来外视界；电磁势中出现的纯 $r$ 项也可由规范变换消去。对亚极端精确解继续作解析延拓，可依次穿过外视界与内视界，并在避开 $\Sigma=0$ 时穿过 $r=0$ 圆盘进入 $r<0$ 区域，继而出现更多渐近平直区域。其径向因果块与亚极端 Reissner--Nordstr\"om 时空同样形成无限链，但 KN 时空不是球对称的：把角方向全部压掉的二维 Penrose 图只能表示这一块结构，不能显示环奇性和穿过圆盘时对 $\theta$ 的依赖。

在最大解析延拓中，$r=r_-$ 是 Cauchy 视界：越过它以后的场不再由某个外部 Cauchy 面上的数据唯一决定。来自外部的微小入射与出射扰动会在此产生无限蓝移；因此理想 KN 解的光滑内视界对扰动不稳定，坍缩黑洞的实际内部一般不会保持上述规则的无限链式结构。

KN 测地线虽不再具有球对称性，却仍因隐藏对称性而完全可分离。对中性测地线记守恒量 $E=-p_t$、$L=p_\phi$，并令 $\epsilon=1$ 表示单位质量类时测地线、$\epsilon=0$ 表示类光测地线。Hamilton--Jacobi 方程给出 Carter 常数 $\mathcal C$；若
\[
    \mathcal P(r)=E(r^2+a^2)-aL,
\]
则一阶方程可写成
\begin{align}
    \Sigma^2\dot r^2&=\mathcal P^2
      -\Delta\bigl[\epsilon r^2+(L-aE)^2+\mathcal C\bigr],\label{eq:kn-radial}\\
    \Sigma^2\dot\theta^2&=\mathcal C-\cos^2\theta
      \left[a^2(\epsilon-E^2)+\frac{L^2}{\sin^2\theta}\right],\label{eq:kn-angular}\\
    \Sigma\dot t&=a(L-aE\sin^2\theta)
      +\frac{r^2+a^2}{\Delta}\mathcal P,\qquad
    \Sigma\dot\phi=\frac{L}{\sin^2\theta}-aE
      +\frac{a}{\Delta}\mathcal P .\label{eq:kn-tphi}
\end{align}
点表示对固有时或仿射参数求导。$\mathcal C$ 控制纬向运动；赤道测地线可取 $\mathcal C=0$。式 \eqref{eq:kn-radial} 与 \eqref{eq:kn-angular} 将允许运动区分别化为两个一维有效势问题，也是研究光子轨道、黑洞阴影和粒子俘获的起点。

\subsection{视界、静止极限与能层}

真正的曲率奇性位于
\[
    \Sigma=0\quad\Longleftrightarrow\quad r=0,
    \quad\theta=\frac{\pi}{2}.
\]
在 Kerr--Schild 空间坐标中它对应 $x^2+y^2=a^2,z=0$ 的环，故称为\textbf{环奇性}(ring singularity)。$r=0$ 的其余部分只是坐标圆盘，可以解析穿过；这与 Schwarzschild 中整个 $r=0$ 边界均为奇性截然不同。

外视界的类光生成元是两个 Killing 矢量的线性组合
\begin{equation}\label{eq:kn-generator}
    \chi=\xi+\Omega_H\psi,
    \qquad \Omega_H=\frac{a}{r_+^2+a^2}.
\end{equation}
$\chi$ 在 $r=r_+$ 上为类光，故外视界是 Killing 视界；$\Omega_H$ 与 $\theta$ 无关，表示整个视界像刚体一样以同一角速度旋转。其空间截面的面积为
\begin{equation}\label{eq:kn-area}
    A_H=4\pi(r_+^2+a^2).
\end{equation}
坐标变换 \eqref{eq:kn-ingoing} 表明视界处曲率正则，而 $\Sigma=0$ 处的曲率不变量发散，二者不可混同。

另一方面，稳态 Killing 矢量 $\xi=\p_t$ 的范数在 $g_{tt}=0$ 处改变符号。外支为
\begin{equation}\label{eq:kn-static-limit}
    r_{\rm s}(\theta)=M+
       \sqrt{M^2-Q^2-a^2\cos^2\theta},
\end{equation}
称为\textbf{外静止极限面}(outer stationary-limit surface)。在旋转轴上 $r_{\rm s}=r_+$，在其他纬度则 $r_{\rm s}>r_+$。夹在 $r_+$ 与 $r_{\rm s}(\theta)$ 之间的区域称为\textbf{能层}(ergoregion)。其中 $\xi$ 是类空的，保持 $r,\theta,\phi$ 全部不变的“静止”世界线不可能为类时；任何物体都必须随黑洞作某种程度的同向旋转。

对静止发射者而言，红移因子 $\sqrt{-g_{tt}}$ 在 $r=r_{\rm s}$ 趋于零，所以静止极限面也常称为\textbf{无限红移面}。不过旋转时一般稳态观者沿 $\xi+\Omega\psi$ 而非 $\xi$ 运动，其红移还依赖 $\Omega$ 与光子的角动量；因此无限红移面不是事件视界。允许的稳态角速度范围由
\begin{equation}\label{eq:kn-omega-range}
    \Omega_-<\Omega<\Omega_+,
    \qquad
    \Omega_\pm=\frac{-g_{t\phi}\pm
      \sqrt{g_{t\phi}^2-g_{tt}g_{\phi\phi}}}{g_{\phi\phi}}
\end{equation}
给出。由于 $g_{t\phi}^2-g_{tt}g_{\phi\phi}=\Delta\sin^2\theta$，趋近外视界时两个根合并为 $\Omega_H$。

轴向 Killing 矢量 $\psi=\p_\phi$ 的轨道因 $\phi\sim\phi+2\pi$ 而闭合，其范数正是 $g_{\phi\phi}$。一旦某处 $g_{\phi\phi}<0$，固定 $t,r,\theta$ 而绕 $\phi$ 一周的曲线便成为闭合类时线。例如在赤道面上
\begin{equation}\label{eq:kn-ctc}
    g_{\phi\phi}\big|_{\theta=\pi/2}
      =r^2+a^2+\frac{2Ma^2}{r}-\frac{a^2Q^2}{r^2}.
\end{equation}
对 $Q\ne0$，最后一项使环奇性附近出现 $g_{\phi\phi}<0$ 的区域；在 Kerr 极限 $Q=0$ 中，相应区域主要出现在穿过圆盘后的负 $r$ 一侧。

亚极端黑洞外部 $r>r_+$ 满足 $g_{\phi\phi}>0$，所以这些显式 CTC 被事件视界乃至内视界遮蔽，并不意味着外部观者可以利用 KN 黑洞回到自己的过去。超极端解没有视界，奇性与因果病态可能裸露，这也是 $M^2<a^2+Q^2$ 不被视作规则黑洞的重要原因。即使在理想最大延拓中，Cauchy 视界的不稳定性也提示负 $r$ 区域和其中的 CTC 不应被直接当作真实坍缩内部的可靠预言。

\subsection{局域非转动观者}

在外部取固定 $r,\theta$ 的观者，并要求其轴向角动量为零：
\[
    u_\phi=g_{t\phi}u^t+g_{\phi\phi}u^\phi=0.
\]
则其相对于无穷远的角速度并不为零，而是
\begin{equation}\label{eq:kn-zamo-omega}
    \omega=\frac{\d\phi}{\d t}
      =-\frac{g_{t\phi}}{g_{\phi\phi}}
      =\frac{a(2Mr-Q^2)}{\mathcal A}.
\end{equation}
这种观者称为\textbf{零角动量观者}(zero-angular-momentum observer, ZAMO)，也称为\textbf{局域非转动观者}。后一个名称表示其空间标架不相对于局域惯性罗盘旋转，并非说它相对于无穷远的 $\phi$ 坐标静止。

归一化后，ZAMO 的 4-速度为
\begin{equation}\label{eq:kn-zamo}
    n=\alpha^{-1}(\p_t+\omega\p_\phi),
    \qquad
    \alpha=\sqrt{\frac{\Delta\Sigma}{\mathcal A}}.
\end{equation}
$\alpha$ 是 Boyer--Lindquist 时间切片的时移。远处 $\omega\sim2J/r^3$，给出弱场的 Lense--Thirring 拖曳；趋近外视界时 $\omega\to\Omega_H$、$\alpha\to0$。因此任何贴近视界而保持 $r,\theta$ 不变的观者都被迫以 $\Omega_H$ 共转，同时需要无限大的加速度；ZAMO 本身不能作为类时观者停在视界上。

\subsection{Penrose 过程}

稳态与轴对称性使中性粒子的
\[
    E=-p_\mu\xi^\mu=-p_t,
    \qquad L=p_\mu\psi^\mu=p_\phi
\]
沿测地线守恒。在能层外，$\xi$ 为未来类时，任意未来因果粒子必有 $E>0$；在能层内 $\xi$ 变为类空，遂允许相对于无穷远具有 $E<0$ 的未来因果轨道。这并非粒子在局域标架中具有负能量，而是 Killing 能量的符号发生了变化。

设能量为 $E_0$ 的粒子进入能层并裂变为两部分，其中一部分以 $E_1<0$ 落入黑洞，另一部分逃向无穷远。由 4-动量守恒，逃逸粒子的能量为
\[
    E_2=E_0-E_1>E_0.
\]
多出的能量来自黑洞的转动，这就是\textbf{Penrose 过程}。负能量碎片的角动量必须大体与黑洞自旋反向；对穿过未来视界的中性粒子，由 $-p\cdot\chi\geqslant0$ 还得到
\begin{equation}\label{eq:kn-horizon-inequality}
    E-\Omega_H L\geqslant0.
\end{equation}
因而黑洞吸收该碎片时 $\delta M=E<0$、$\delta J=L<0$，并满足 $\delta M-\Omega_H\delta J\geqslant0$。

定义\textbf{不可约质量}
\begin{equation}\label{eq:kn-irreducible-mass}
    M_{\rm irr}=\sqrt{\frac{A_H}{16\pi}}
      =\frac12\sqrt{r_+^2+a^2},
\end{equation}
则 KN 参量满足 Christodoulou--Ruffini 关系
\begin{equation}\label{eq:kn-cr-formula}
    M^2=\left(M_{\rm irr}+\frac{Q^2}{4M_{\rm irr}}\right)^2
       +\frac{J^2}{4M_{\rm irr}^2}.
\end{equation}
在满足通常能量条件的经典过程中，视界面积因而 $M_{\rm irr}$ 不能减小，可提取的是式 \eqref{eq:kn-cr-formula} 中的转动能与适当条件下的电磁能。对极端 Kerr 黑洞，理想单次裂变的最大增益约为入射能量的 $20.7\%$；若以一系列近可逆过程逐渐去旋，理论上最多可提取初始质量的 $1-1/\sqrt2\simeq29\%$。天体物理中更有效的电磁提能机制虽不等同于原始 Penrose 裂变，其几何根源同样是能层和视界旋转。

\subsection{黑洞无毛}

“黑洞无毛”是若干唯一性定理的概括，而不是单独一条不附条件的定理。在四维 Einstein--Maxwell 理论中，若时空渐近平直、稳态，域外区规则且具有适当的全局条件，并假定连通的非退化事件视界以及定理所需的解析性或相应正则性，则黑洞外部由 $M,J,Q$ 唯一确定，即与某个 Kerr--Newman 解等距。若允许磁单极，还须再加入磁荷 $P$，并将许多公式中的 $Q^2$ 换成 $Q^2+P^2$。

“无毛”并不是说外场没有高阶多极矩，而是说这些多极矩不是独立参量。以适当的质量多极矩 $M_\ell$、流多极矩 $S_\ell$ 以及电、磁多极矩 $Q_\ell,P_\ell$ 表示，KN 解满足
\begin{equation}\label{eq:kn-multipoles}
    M_\ell+\mathrm{i}S_\ell=M(\mathrm{i}a)^\ell,
    \qquad
    Q_\ell+\mathrm{i}P_\ell=Q(\mathrm{i}a)^\ell .
\end{equation}
例如 $M_0=M$、$S_1=J$、$M_2=-Ma^2$；一旦测得 $M,J,Q$，所有更高矩都已固定。这使观测黑洞四极矩成为检验 Kerr--Newman 几何的重要思路。

唯一性定理也不自动证明任意坍缩解必在晚时趋于 KN 解。后者还涉及辐射衰减、视界稳定性和物质清空等动力学问题。若改变渐近条件、加入外部物质，或改用含额外标量场、非 Abel 规范场及高曲率项的引力理论，黑洞可以具有额外参量甚至真正的“毛”。因此无毛结论必须连同其理论、边界条件和正则性假设一起使用。

\section{黑洞热力学}\label{sec:bh-thermodynamics}

黑洞在经典理论中是单向的因果边界，却具有一套与热力学惊人平行的定律：视界面积对应熵，表面引力对应温度，质量则对应能量。这种对应并非纯粹形式上的类比。量子场论预言黑洞以 Hawking 温度辐射，而其熵正比于视界面积，因而将引力、量子理论和统计物理联系在一起。本节仍取 $G=c=k_B=1$，但保留 $\hbar$，以区分经典结论与量子结论。

\subsection{传统框架及其局限}

传统讨论以渐近平直的稳态时空和事件视界为核心：Killing 对称性给出质量、角动量、表面引力等守恒量，Raychaudhuri 方程导出面积不减定理，而相邻稳态解之间满足第一定律。这一框架最适合 Kerr--Newman 黑洞及其微扰，并自然导向“面积像熵、表面引力像温度”的四定律。Hawking 辐射进一步赋予二者真实的温度与熵，使经典黑洞力学成为半经典黑洞热力学。

不过，事件视界 $\mathcal H^+=\partial J^-(\mathscr I^+)$ 是\textbf{遥感的}(teleological)：必须知道时空的完整未来才能判定它的位置。它可以为响应将来才落入的物质而在局部真空甚至平直区域中提前扩张，因此面积增量不能普遍解释为当时穿过事件视界的局域能流。它还依赖未来类光无穷远，在空间紧致时空中不能照搬；在数值演化中，也无法只由一个时刻的初始数据立即找到。

稳态假设同样过强：黑洞附近已经平衡时，远处仍可能存在物质或引力辐射；传统第一定律还把无穷远定义的 $M,J$ 与视界上的 $\kappa,\Omega_H$ 混在一起，难以单独描述“黑洞自身”的能量收支。表观视界虽能由单个空间切片求得，却依赖切片选择。由此需要一组层次分明的准局域概念：表观视界用于逐时刻定位，弱孤立视界描述局域平衡，动力学视界描述吸积和并合中的有限通量。

\subsection{Killing 视界与表面引力}

对稳态黑洞，事件视界通常同时是某个 Killing 矢量 $\chi^a$ 的 Killing 视界。也就是说，$\chi^a$ 在视界上既与视界相切又为类光，并生成其类光母线。对 Kerr--Newman 黑洞，式 \eqref{eq:kn-generator} 已给出 $\chi=\xi+\Omega_H\psi$。沿视界生成元的参数一般不是仿射参数，而满足
\begin{equation}\label{eq:surface-gravity-def}
    \chi^b\nabla_b\chi^a=\kappa\chi^a,
    \qquad
    \nabla_a(\chi^b\chi_b)=-2\kappa\chi_a
    \quad\text{于 }\mathcal H^+ .
\end{equation}
比例系数 $\kappa$ 称为\textbf{表面引力}(surface gravity)。在渐近平直时空中，把 $\xi$ 归一化为无穷远处的单位时间平移便固定了 $\chi$ 的尺度，因而也固定了 $\kappa$。对亚极端 Kerr--Newman 黑洞，
\begin{equation}\label{eq:kn-thermo-potentials}
    \kappa=\frac{r_+-r_-}{2(r_+^2+a^2)},\qquad
    \Omega_H=\frac{a}{r_+^2+a^2},\qquad
    \Phi_H=\frac{Qr_+}{r_+^2+a^2}.
\end{equation}
这里 $\Phi_H$ 是视界生成元所测得的电势与无穷远电势之差，其符号随电磁势和电荷的约定一起改变。极端极限 $r_+=r_-$ 给出 $\kappa=0$。

表面引力也可由贴近视界且沿 $\chi$ 运动的稳态观者理解。令 $V=\sqrt{-\chi^2}$，则这类观者的固有加速度在视界处发散，但按 Killing 时间红移后的量 $Va$ 趋于有限极限 $\kappa$；$\chi$ 的尺度仍由无穷远的时间平移固定。因此 $\kappa$ 是红移后的“视界加速度”，而不是自由落体观者在视界处测得的局域力。

在 Einstein--Maxwell 理论并满足适当能量条件时，稳态 Killing 视界上的 $\kappa$ 为常数，这就是黑洞力学的\textbf{第零定律}。其证明本质上利用 Killing 恒等式、场方程以及视界生成元无膨胀、无剪切的性质；它与平衡态温度在整个系统中均匀的热力学第零定律相对应。





\subsection{黑洞力学四定律}

对相邻的稳态 Kerr--Newman 解作变分，可得黑洞力学\textbf{第一定律}
\begin{equation}\label{eq:bh-first-law}
    \delta M=\frac{\kappa}{8\pi}\,\delta A_H
       +\Omega_H\,\delta J+\Phi_H\,\delta Q .
\end{equation}
$\Omega_H$ 和 $\Phi_H$ 分别是角动量与电荷的化学势。该式既可看作稳态解空间中的参数恒等式，也有“物理过程”版本：少量物质和辐射落入黑洞、扰动最终衰减后，能量、角动量和电荷通量引起的面积变化仍满足同一关系。对有限的稳态 KN 解，还存在 Smarr 关系
\begin{equation}\label{eq:bh-smarr}
    M=\frac{\kappa A_H}{4\pi}+2\Omega_HJ+\Phi_HQ,
\end{equation}
它可由尺度齐次性或 Komar 积分推出，不能简单地由第一定律把微分符号删去得到。

四条定律可概括如下。第零定律说平衡视界上的 $\kappa$ 为常数；第一定律是式 \eqref{eq:bh-first-law}；第二定律即经典面积不减定理；第三定律的不可达表述则说，不能用满足适当正则性和能量条件的有限物理过程把非极端黑洞降到 $\kappa=0$。最后一条必须连同其过程假设理解，不能仅凭极端 KN 解本身存在就判为错误。

在量子效应尚未引入时，$\kappa$ 与 $A_H$ 只是分别类似温度与熵的几何量：经典黑洞不发射热辐射，面积定理又依赖能量条件而不是统计力学。因此以上结论严格说是\textbf{黑洞力学定律}。Hawking 辐射给出 $T_H$，第一定律随即固定 $S_{\rm BH}$，这才把形式类比提升为\textbf{黑洞热力学定律}。对于动力学视界，下文得到的是有限过程中的几何通量和平衡律；是否、以及怎样把它解释成远离平衡态的温度和熵，仍须另加半经典与统计物理输入。

面积定理是经典结论，依赖能量条件和整体因果假设；Hawking 辐射中的负能流会破坏其经典前提，使蒸发黑洞的面积可以减小。

\subsection{Hawking 辐射与 Bekenstein--Hawking 熵}

在形成黑洞的坍缩背景上量子化物质场时，抵达未来类光无穷远的出射模在向过去追溯时会受到指数红移。正、负频率模因此发生混合，使无穷远观者看到近似热谱。对稳态黑洞，其温度为
\begin{equation}\label{eq:hawking-temperature}
    T_H=\frac{\hbar\kappa}{2\pi}.
\end{equation}
同一结果也可由 Euclidean 方法看出：视界附近的二维度规具有 Rindler 形式，令 $t=-\mathrm{i}\tau$ 后，要消除 Euclidean 截面原点处的锥形奇性，必须令 $\tau$ 具有周期 $2\pi/\kappa$；这正对应逆温度 $\beta=\hbar/T_H$。

无穷远处的谱并非理想黑体谱，因为波在离开黑洞时还要穿过由时空曲率产生的有效势垒。各角动量模的透射率称为\textbf{灰体因子}(greybody factor)。旋转或带电黑洞的辐射还含有化学势 $\Omega_H$ 和 $\Phi_H$；正频玻色模在 $0<\omega<m\Omega_H+q\Phi_H/\hbar$ 的范围内表现出超辐射放大。温度为零的极端极限需要单独处理，不能把非极端态的晚时推导不加分析地直接取极限。

将式 \eqref{eq:hawking-temperature} 与第一定律比较，便确定黑洞熵为
\begin{equation}\label{eq:bek-hawking-entropy}
    S_{\rm BH}=\frac{A_H}{4\hbar}.
\end{equation}
这就是\textbf{Bekenstein--Hawking 熵}。面积而不是体积控制熵，是量子引力最重要的线索之一。对 Schwarzschild 黑洞，$A_H=16\pi M^2$ 且 $\kappa=1/(4M)$，所以
\begin{equation}\label{eq:schwarzschild-thermo}
    T_H=\frac{\hbar}{8\pi M},\qquad
    S_{\rm BH}=\frac{4\pi M^2}{\hbar},\qquad
    C=\dv{M}{T_H}=-\frac{8\pi M^2}{\hbar}.
\end{equation}
负热容表示孤立黑洞越辐射越热。在渐近平直真空中，其功率量级为 $P\sim\hbar/M^2$，寿命量级为 $t_{\rm evap}\sim M^3/\hbar$；数值系数取决于可辐射粒子种类和灰体因子。负热容也说明渐近平直黑洞不能像普通物质那样与无限大热库形成稳定正则系综；若置于有限空腔或 AdS 时空，稳定性则可能改变。

\subsection{广义第二定律}

若把普通熵携带进黑洞，外部可见物质的熵会下降。Bekenstein 的关键提议是把视界熵计入总账，定义\textbf{广义熵}
\begin{equation}\label{eq:generalized-entropy}
    S_{\rm gen}=\frac{A_H}{4\hbar}+S_{\rm out},
\end{equation}
其中 $S_{\rm out}$ 是视界外量子场的熵。\textbf{广义第二定律}断言在适用的半经典过程中 $\Delta S_{\rm gen}\geqslant0$。物质落入时 $S_{\rm out}$ 可减小，但面积项增加；Hawking 蒸发时面积项减小，辐射熵则增加。二者共同取代单独的经典面积定理。

$S_{\rm out}$ 的纠缠熵在视界附近含紫外发散，面积项所用的 Newton 常数也随重整化尺度变化；两部分分别依赖正则化方案，而适当重整化后的 $S_{\rm gen}$ 才具有物理意义。半经典展开还允许对面积律加入对数项等量子修正，但在 $A_H\gg\hbar$ 时，式 \eqref{eq:bek-hawking-entropy} 是主导项。

\subsection{非膨胀、弱孤立与孤立视界}

为了描述“黑洞本身已平衡、远处仍可有辐射”的情形，可用局域视界取代全局 Killing 视界。第一层结构是\textbf{非膨胀视界}(non-expanding horizon, NEH)：$\Delta$ 是拓扑为 $S^2\times\mathbb R$ 的类光超曲面，其退化内禀度规为 $q_{ab}$，任一未来类光法矢 $\ell^a$ 都满足 $\theta_{(\ell)}=0$，并要求场方程及适当的主导能量条件在 $\Delta$ 上成立。Raychaudhuri 方程随即给出
\[
    \sigma^{(\ell)}_{ab}=0,\qquad
    T_{ab}\ell^a\ell^b=0,\qquad
    \mathcal L_\ell q_{ab}=0.
\]
所以各截面面积相同，内禀度规沿生成元不变，也没有物质或引力辐射穿过 $\Delta$。

NEH 上的时空联络诱导出唯一的无挠内禀联络 $\mathcal D_a$。取仅相差正常数倍的法矢等价类 $[\ell]=\{c\ell:c>0\}$，并以
\[
    \mathcal D_a\ell^b=\omega_a\ell^b
\]
定义法丛联络 1-形式 $\omega_a$。若进一步要求
\begin{equation}\label{eq:wih-condition}
    [\mathcal L_\ell,\mathcal D_a]\ell^b=0,
    \qquad\text{等价地}\qquad
    \mathcal L_\ell\omega_a=0,
\end{equation}
则 $(\Delta,[\ell])$ 称为\textbf{弱孤立视界}(weakly isolated horizon, WIH)。其表面引力为 $\kappa_{(\ell)}=\ell^a\omega_a$。在视界上 $\iota_\ell\d\omega=0$，故由 Cartan 恒等式 $\mathcal L_\ell\omega=\d\kappa_{(\ell)}+\iota_\ell\d\omega$ 与式 \eqref{eq:wih-condition} 得 $\d\kappa_{(\ell)}=0$；这就是不依赖外部全局稳态的\textbf{第零定律}。重标度 $\ell\mapsto c\ell$ 会令 $\kappa_{(\ell)}\mapsto c\kappa_{(\ell)}$，但不破坏其在视界上的常值性。

若把式 \eqref{eq:wih-condition} 加强为
\[
    [\mathcal L_\ell,\mathcal D_a]X^b=0
    \quad\text{对任意切于 }\Delta\text{ 的 }X^a,
\]
就得到\textbf{孤立视界}(isolated horizon, IH)：不仅 $q_{ab}$ 和 $\mathcal D_a\ell^b$，整个内禀联络都沿 $\ell^a$ 不变。这里的“孤立”是视界内禀几何的对称性，并不要求 $\ell^a$ 延拓为邻域乃至全时空的 Killing 场。IH 的对称保持 $(q_{ab},\mathcal D_a,[\ell])$，WIH 的对称保持 $(q_{ab},\omega_a,[\ell])$。球形截面上由此有三类：I 型球对称，对称群为四维；II 型轴对称，对称群为二维；III 型除生成元方向外无连续对称，对称群为一维。

在 Hamilton 框架中，允许的视界时间演化存在，当且仅当相应的视界能量满足
\begin{equation}\label{eq:isolated-horizon-first-law}
    \delta E_\Delta=\frac{\kappa_{(\ell)}}{8\pi}\delta A_\Delta
      +\Omega_\Delta\delta J_\Delta+\Phi_\Delta\delta Q_\Delta .
\end{equation}
这给出 WIH 的\textbf{第一定律}，其中各量均可在视界上准局域定义。不过没有无穷远归一化时，$E_\Delta$ 和 $\kappa_{(\ell)}$ 并非自动唯一；必须连同所选的视界时间演化一起固定。

\subsection{动力学视界}

正在吸积或并合的黑洞不能由非膨胀视界描述。\textbf{动力学视界}(dynamical horizon, DH)是可由闭合 MOTS 分层的类空 3-曲面 $H$，每片 $S$ 满足 $\theta_{(\ell)}=0$、$\theta_{(n)}<0$。因为 $H$ 是类空的，物质和引力辐射可以穿过它；在通常能量条件下，其叶片面积沿外向方向严格增加，并在通量消失时趋向类光的孤立视界。以下取 $\Lambda=0$ 且叶片拓扑为 $S^2$。

令 $\hat\tau^a$ 是 $H$ 的未来单位类时法矢，$h_{ab}=g_{ab}+\hat\tau_a\hat\tau_b$ 是 $H$ 的三维诱导度规，$K_{ab}=h_a{}^ch_b{}^d\nabla_c\hat\tau_d$ 是外曲率。再令 $\hat r^a$ 为 $H$ 内垂直于叶片的外向单位法矢，$q_{ab}=h_{ab}-\hat r_a\hat r_b$ 为 $S$ 的二维度规，$\widetilde K_{ab}=q_a{}^cq_b{}^dD_c\hat r_d$ 为 $S\subset H$ 的外曲率。固定
\[
    \ell^a=\hat\tau^a+\hat r^a,\qquad
    n^a=\hat\tau^a-\hat r^a,\qquad
    \ell\cdot n=-2,
\]
则分层条件与几何量的关系为
\begin{equation}\label{eq:dh-foliation-geometry}
    \theta_{(\ell)}=q^{ab}(K_{ab}+\widetilde K_{ab})=0,
    \qquad
    \theta_{(n)}=q^{ab}(K_{ab}-\widetilde K_{ab})<0.
\end{equation}
特别地，$\widetilde K=q^{ab}D_a\hat r_b=-\theta_{(n)}/2>0$，所以叶片面积沿 $\hat r^a$ 增加。三维数据 $(h_{ab},K_{ab})$ 还满足 Hamilton 与动量约束
\begin{equation}\label{eq:dh-constraints}
    {}^{(3)}R+K^2-K_{ab}K^{ab}
       =16\pi T_{ab}\hat\tau^a\hat\tau^b,
    \qquad
    D_b(K^{ab}-Kh^{ab})
       =8\pi T^{bc}\hat\tau_c h^a{}_b .
\end{equation}
正是这些约束经 $2+1$ 分解后产生下列通量平衡律。

令叶片面积为 $A=4\pi R^2$，$N_R=|D R|$、$\xi^a_{(R)}=N_R\ell^a$，并定义 $\zeta_a=q_a{}^b\hat r^c\nabla_c\ell_b$。两个叶片之间的\textbf{面积平衡定律}为
\begin{equation}\label{eq:dynamical-horizon-balance}
\begin{split}
    \frac{R_2-R_1}{2}={}&
    \int_{\Delta H}T_{ab}\hat\tau^a\xi^b_{(R)}\,\d^3V\\
    &+\frac{1}{16\pi}\int_{\Delta H}N_R
      \left(\sigma^{(\ell)}_{ab}\sigma_{(\ell)}^{ab}
      +2\zeta_a\zeta^a\right)\d^3V .
\end{split}
\end{equation}
第一项是物质能流，第二项是由剪切和横向法丛几何产生的非负引力能流。在无转动情形取 $E_H=R/2$、$\kappa_{\rm eff}=1/(2R)$，式 \eqref{eq:dynamical-horizon-balance} 的无穷小形式就是
\begin{equation}\label{eq:dh-nonrotating-first-law}
    \d E_H=\frac{\kappa_{\rm eff}}{8\pi}\d A ,
\end{equation}
即角动量为零的第一定律。

若叶片具有轴向矢量 $\varphi^a$，可定义准局域角动量
\begin{equation}\label{eq:dh-angular-momentum}
    J_S^{(\varphi)}
      =-\frac{1}{8\pi}\oint_S K_{ab}\varphi^a\hat r^b\,\d^2V .
\end{equation}
记 $P^{ab}=K^{ab}-Kh^{ab}$，动量约束给出\textbf{角动量平衡方程}
\begin{equation}\label{eq:dh-angular-momentum-balance}
\begin{split}
    J_{S_2}^{(\varphi)}-J_{S_1}^{(\varphi)}
      ={}&-\int_{\Delta H}T_{ab}\hat\tau^a\varphi^b\,\d^3V\\
       &-\frac{1}{16\pi}\int_{\Delta H}
          P^{ab}\mathcal L_\varphi h_{ab}\,\d^3V .
\end{split}
\end{equation}
右边两项分别是物质与引力的角动量通量；若 $\varphi^a$ 是整个三维度规 $h_{ab}$ 的精确 Killing 场，第二项为零。

为显示有限过程的第一定律，先忽略电荷并选取与 Kerr 关系相容的视界能量与势
\[
    E_H(R,J)=\frac{\sqrt{R^4+4J^2}}{2R},\qquad
    \kappa_{\rm eff}=\frac{R^4-4J^2}
      {2R^3\sqrt{R^4+4J^2}},\qquad
    \Omega_{\rm eff}=\frac{2J}{R\sqrt{R^4+4J^2}}.
\]
于是相邻叶片满足 $\d E_H=\kappa_{\rm eff}\d A/(8\pi)+\Omega_{\rm eff}\d J$。令 $r^0(R)$ 满足 $\dv*{r^0}{R}=2R\kappa_{\rm eff}$，并记 $N_{r^0}=|D r^0|$；沿 $\Delta H$ 积分则得到
\begin{equation}\label{eq:dh-integral-first-law}
\begin{split}
    E_H(S_2)-E_H(S_1)
      &=
      \int_{S_1}^{S_2}
      \left(\frac{\kappa_{\rm eff}}{8\pi}\d A
      +\Omega_{\rm eff}\d J\right)\\
      &=\int_{\Delta H}T_{ab}\hat\tau^at^b\,\d^3V
      +\frac{1}{16\pi}\int_{\Delta H}N_{r^0}
        \left(\sigma^{(\ell)}_{ab}\sigma_{(\ell)}^{ab}
        +2\zeta_a\zeta^a\right)\d^3V ,
\end{split}
\end{equation}
其中 $t^a=N_{r^0}\ell^a+\Omega_{\rm eff}\varphi^a$，$N_{r^0}$ 由上述 $\kappa_{\rm eff}$ 固定。式 \eqref{eq:dh-integral-first-law} 是允许初末叶片相距有限的\textbf{第一定律积分形式}；它描述同一时空中的真实通量过程，而不是相邻稳态解之间的参数变分。带电情形还需加入 $\Phi_{\rm eff}\d Q$ 及相应的电磁能流。

\subsection{信息论、量子与黑洞}

量子信息论用密度算符 $\rho$ 描述一个系统，并以 von Neumann 熵
\[
    S(\rho)=-\Tr(\rho\ln\rho)
\]
量度其与未观测自由度的纠缠。封闭系统的幺正演化保持全系统的精细熵；但对子系统取偏迹后，约化态可以是混态并具有非零熵。因此讨论“信息是否丢失”时，必须区分整个引力系统、外部辐射和黑洞内部，也要区分精细纠缠熵与忽略微观关联后的粗粒化热力学熵。

Hawking 的原始半经典计算给出的晚时辐射近似热化；每个外部 Hawking 量子都与视界内的伙伴纠缠。若黑洞完全蒸发，而外部辐射中只留下热关联，一个初始纯态似乎会演化为混态，与封闭量子系统的幺正演化冲突。这就是\textbf{黑洞信息问题}，也是黑洞热力学从宏观平衡律走向\textbf{黑洞信息学}的核心问题。

若演化幺正，收集到的 Hawking 辐射熵应先增长，在蒸发过程的中段达到最大值，再下降至零；这条预期曲线称为 Page 曲线。固定背景上的局域半经典计算却让外部辐射不断与内部伙伴纠缠，因而只得到持续上升的熵。两者的冲突表明，在蒸发后期不能同时坚持半经典局域性、光滑视界和无残余信息丢失等全部假设。

引力路径积分中的\textbf{岛公式}为辐射区域 $R$ 的熵给出
\begin{equation}\label{eq:island-formula}
    S(R)=\min_I\underset{\partial I}{\operatorname{ext}}
    \left[\frac{\operatorname{Area}(\partial I)}{4\hbar}
      +S_{\rm matter}(R\cup I)\right],
\end{equation}
其中 $I$ 是可纳入辐射纠缠楔的“岛”。在许多可控模型中，早期由无岛鞍点主导，晚期则由含岛鞍点主导，从而重现 Page 曲线。这说明半经典引力已经编码了保持总熵有限的重要机制，但它本身并未提供任意四维坍缩中微观信息如何被解码的完整动力学图景。黑洞熵的微观态计数、视界局域性以及蒸发终态仍是量子引力理论必须共同回答的问题。


\section{dS 与 AdS 时空}\label{sec:dS}

含宇宙学常数的真空 Einstein 方程并不只允许平直解。正宇宙学常数给出 de Sitter 时空，负宇宙学常数给出 anti-de Sitter 时空；它们分别简称 dS 与 AdS。两者的局域曲率都极其简单，却有完全不同的共形边界和因果结构，因而也是理解宇宙学视界、黑洞热力学及全息思想的基本背景。

\subsection{常曲率空间}

四维无物质场方程为
\begin{equation}\label{eq:einstein-lambda-vacuum}
    R_{\mu\nu}-\frac12 R g_{\mu\nu}+\Lambda g_{\mu\nu}=0.
\end{equation}
取迹后得到 $R=4\Lambda$，再代回原式即有 $R_{\mu\nu}=\Lambda g_{\mu\nu}$。不过，Ricci 张量与度规成比例只说明时空是 Einstein 空间，尚不自动保证其截面曲率处处相同；dS 与 AdS 还满足更强的最大对称性。定义曲率半径 $L>0$ 及符号 $\epsilon$ 为
\[
    \Lambda=\epsilon\frac{3}{L^2},\qquad
    \epsilon=+1\ \text{对应 dS},\qquad
    \epsilon=-1\ \text{对应 AdS},
\]
则其完整曲率张量为
\begin{equation}\label{eq:maximally-symmetric-curvature}
    R_{\mu\nu\rho\sigma}
      =\frac{\epsilon}{L^2}
       \left(g_{\mu\rho}g_{\nu\sigma}-g_{\mu\sigma}g_{\nu\rho}\right),
    \qquad
    R_{\mu\nu}=\frac{3\epsilon}{L^2}g_{\mu\nu},
    \qquad
    R=\frac{12\epsilon}{L^2}.
\end{equation}
因此 Weyl 张量为零，且
\[
    R_{\mu\nu}R^{\mu\nu}=\frac{36}{L^4},\qquad
    R_{\mu\nu\rho\sigma}R^{\mu\nu\rho\sigma}=\frac{24}{L^4}.
\]
这些不变量处处有限，故 dS、AdS 的坐标边界都不能仅凭曲率发散判作奇性。

四维最大对称时空具有 $4\times5/2=10$ 个独立 Killing 矢量。Minkowski、dS 与 AdS 分别是零、正、负常曲率的 Lorentz 时空；忽略离散分支与万有覆盖的差别，其等距群分别为 Poincar\'e 群 $ISO(1,3)$、$SO(1,4)$ 与 $SO(2,3)$。若把宇宙学常数移到方程右侧，它等效于
\[
    T^{(\Lambda)}_{\mu\nu}=-\frac{\Lambda}{8\pi}g_{\mu\nu},
    \qquad
    \rho_\Lambda=\frac{\Lambda}{8\pi},\qquad
    p_\Lambda=-\rho_\Lambda,
\]
即状态方程 $p=-\rho$ 的 Lorentz 不变真空能。

\subsection{dS 时空}

在五维平直空间 $\mathbb R^{1,4}$ 中取度规
$\d s_5^2=-(\d X^0)^2+\sum_{A=1}^4(\d X^A)^2$。单叶双曲面
\begin{equation}\label{eq:dS-embedding}
    -(X^0)^2+\sum_{A=1}^4(X^A)^2=L^2
\end{equation}
的诱导度规就是四维 \textbf{de Sitter 时空}。令 $n^A$ 是单位 3-球上的坐标，即 $\sum_A(n^A)^2=1$，并取
\[
    X^0=L\sinh(T/L),\qquad
    X^A=L\cosh(T/L)n^A,
\]
便得到覆盖整个双曲面的全局坐标
\begin{equation}\label{eq:dS-global}
    \d s^2=-\d T^2+L^2\cosh^2(T/L)\,\d\Omega_3^2,
    \qquad -\infty<T<\infty.
\end{equation}
每个 $T$ 常数面都是 $S^3$，故整体拓扑为 $\mathbb R\times S^3$。该坐标把 dS 写成闭合 FLRW 宇宙：空间先收缩，在 $T=0$ 达到最小半径 $L$，再重新膨胀。这只是所选分层的表现，并不破坏时空本身的最大对称性。

为看清因果结构，定义全局共形时间
\[
    \tan\eta=\sinh(T/L),\qquad
    -\frac\pi2<\eta<\frac\pi2.
\]
将 $\d\Omega_3^2=\d\chi^2+\sin^2\chi\,\d\Omega^2$ 写成极坐标形式后，式 \eqref{eq:dS-global} 成为
\begin{equation}\label{eq:dS-conformal}
    \d s^2=\frac{L^2}{\cos^2\eta}
      \left(-\d\eta^2+\d\chi^2+\sin^2\chi\,\d\Omega^2\right),
    \qquad 0\leqslant\chi\leqslant\pi.
\end{equation}
去掉共形因子后，径向光线满足 $\d\chi=\pm\d\eta$。过去与未来无穷远 $\mathscr I^-$、$\mathscr I^+$ 分别位于 $\eta=-\pi/2$、$\eta=\pi/2$，都是类空边界；这与渐近平直时空的类光无穷远截然不同。

图 \ref{fig:dS-penrose} 显示，一个从 $\mathscr I^-$ 延伸到 $\mathscr I^+$ 的观者不能与整个 dS 时空交换信号。相对于 $\chi=0$ 的观者 $\gamma$，其过去、未来宇宙学视界在球对称商空间中围出一个静态区域。相应的嵌入坐标变换为
\[
    X^0=\sqrt{L^2-r^2}\sinh(t/L),\qquad
    X^4=\sqrt{L^2-r^2}\cosh(t/L),\qquad
    (X^1,X^2,X^3)=r\bm n,
\]
静态坐标下的度规为
\begin{equation}\label{eq:dS-static}
    \d s^2=-\left(1-\frac{r^2}{L^2}\right)\d t^2
      +\left(1-\frac{r^2}{L^2}\right)^{-1}\d r^2+r^2\d\Omega^2,
    \qquad 0\leqslant r<L.
\end{equation}
$r=L$ 是静态 Killing 矢量 $\partial_t$ 的 Killing 视界，并非曲率奇性。它的表面引力、Gibbons--Hawking 温度和视界熵分别为
\begin{equation}\label{eq:dS-thermodynamics}
    \kappa_{\rm dS}=\frac1L,
    \qquad T_{\rm dS}=\frac{\hbar}{2\pi L},
    \qquad S_{\rm dS}=\frac{4\pi L^2}{4\hbar}=\frac{\pi L^2}{\hbar}.
\end{equation}
这里的视界依赖观者：换一条测地观者世界线会改变静态区域的位置，但不会改变其半径和热力学量。

dS 还可写成空间平直的膨胀坐标
\begin{equation}\label{eq:dS-flat-slicing}
    \d s^2=-\d\tau^2+\e^{2\tau/L}\d\bm x^2
      =\frac{L^2}{\eta_{\rm f}^2}
       \left(-\d\eta_{\rm f}^2+\d\bm x^2\right),
    \qquad \eta_{\rm f}=-L\e^{-\tau/L}<0.
\end{equation}
此时 Hubble 参数恒为 $H=1/L$。这套坐标只覆盖完整 dS 的一个膨胀半部；$\tau\to-\infty$ 的坐标边界不是式 \eqref{eq:dS-embedding} 上的曲率奇性。

\subsection{再谈事件视界和粒子视界}

宇宙学中常把事件视界与粒子视界并列讨论。考虑共动观者位于 $\chi=0$ 的 FLRW 度规
\begin{equation}\label{eq:flrw-conformal-radial}
    \d s^2=a^2(\eta)
      \left[-\d\eta^2+\d\chi^2+S_k^2(\chi)\d\Omega^2\right],
    \qquad \d\eta=\frac{\d t}{a(t)},
\end{equation}
其中 $S_k(\chi)=\sin\chi,\chi,\sinh\chi$ 分别对应 $k=+1,0,-1$。径向光线仍满足 $\d\chi=\pm\d\eta$。若该宇宙学区域的初、末时刻为 $t_i,t_f$，则时刻 $t$ 的两个共动视界尺度为
\begin{equation}\label{eq:particle-event-horizon}
    \chi_{\rm p}(t)=\int_{t_i}^{t}\frac{\d t'}{a(t')}
      =\eta(t)-\eta_i,
    \qquad
    \chi_{\rm e}(t)=\int_t^{t_f}\frac{\d t'}{a(t')}
      =\eta_f-\eta(t).
\end{equation}
第一式给出\textbf{粒子视界}(particle horizon)：超出它的共动源自初始边界以来还没有足够时间把信号送到观者。第二式给出\textbf{宇宙学事件视界}(cosmological event horizon)：现在从更远处发出的信号即使等待到时空的未来端点也不能到达观者。相应积分发散时，该类非平凡视界便不存在；在闭合空间中还须把结果与空间的最大共动距离比较。

粒子视界依赖所选择的初始边界或宇宙学模型，回答的是“至今已经看见什么”；事件视界则依赖完整未来，回答的是“最终还能看见什么”。后者与黑洞事件视界一样具有遥感性，不能由某一时刻的局域测量独立确定。在空间平直的 FLRW 描述中，Hubble 半径 $H^{-1}$ 只是退行速度等于光速的瞬时尺度，一般既不是粒子视界也不是事件视界。

对式 \eqref{eq:dS-flat-slicing} 的永恒膨胀区域，$a(\tau)=\e^{H\tau}$ 且未来端点为 $\tau_f=+\infty$，所以
\[
    \chi_{\rm e}(\tau)=\frac{\e^{-H\tau}}{H},
    \qquad D_{\rm e}=a(\tau)\chi_{\rm e}(\tau)=H^{-1}=L.
\]
故 dS 的事件视界恰与 Hubble 半径重合；这是指数膨胀的特殊性质。若从有限初始时刻开始考察，粒子视界也随初始面的选取而出现。完整全局 dS 中，静态观者同时具有图 \ref{fig:dS-penrose} 所示的过去与未来宇宙学视界。

\subsection{Schwarzschild-dS 时空}

含宇宙学常数的 Birkhoff 定理给出的球对称真空解称为 \textbf{Schwarzschild--de Sitter 时空}或 Kottler 时空：
\begin{equation}\label{eq:schwarzschild-dS}
    \d s^2=-f(r)\d t^2+\frac{\d r^2}{f(r)}+r^2\d\Omega^2,
    \qquad
    f(r)=1-\frac{2M}{r}-\frac{r^2}{L^2},
    \qquad L^2=\frac3\Lambda.
\end{equation}
当 $M=0$ 时它退化为 dS 静态坐标；当 $\Lambda\to0$ 时，宇宙学视界移向无穷远并恢复 Schwarzschild 解。

若
\begin{equation}\label{eq:sdS-horizon-condition}
    0<9M^2\Lambda<1,
\end{equation}
则 $f(r)=0$ 有两个正根 $r_b<r_c$。$r_b$ 是黑洞视界，$r_c$ 是宇宙学视界，而 $r_b<r<r_c$ 是 $\partial_t$ 为类时的静态区域。小 $M/L$ 极限下
\[
    r_b=2M+O(M^3/L^2),\qquad
    r_c=L-M+O(M^2/L).
\]
两个视界重合的极限满足
\[
    9M^2\Lambda=1,\qquad r_b=r_c=\frac1{\sqrt\Lambda}=3M,
\]
称为 \textbf{Nariai 极限}。此时原静态坐标区间缩并；适当放大两视界之间的区域后得到 $\mathrm{dS}_2\times S^2$，而不是简单地把 $r_b=r_c$ 代入后丢弃该区域。

在式 \eqref{eq:schwarzschild-dS} 所选的 Killing 时间归一化下，任一非退化视界 $r_h$ 的表面引力为
\begin{equation}\label{eq:sdS-surface-gravity}
    \kappa_h=\frac12|f'(r_h)|
      =\left|\frac{M}{r_h^2}-\frac{r_h}{L^2}\right|,
    \qquad T_h=\frac{\hbar\kappa_h}{2\pi}.
\end{equation}
一般有 $\kappa_b\ne\kappa_c$，所以同一静态区域的两个视界具有不同温度，不能用一个 Euclidean 时间周期同时消除两端的锥形奇性。又因该时空没有渐近平直无穷远，$\partial_t$ 缺少由无穷远单位观者给出的唯一归一化；讨论绝对温度时必须明确所用的静态观者或归一化约定。

$M$ 所产生的 Weyl 曲率使
\begin{equation}\label{eq:sdS-kretschmann}
    R_{\mu\nu\rho\sigma}R^{\mu\nu\rho\sigma}
      =\frac{48M^2}{r^6}+\frac{8\Lambda^2}{3}.
\end{equation}
因此两支视界都只是坐标奇性，而 $r=0$ 仍是真实曲率奇性。最大延拓包含黑洞、白洞、静态区域和越过宇宙学视界的动力学区域，并沿空间方向形成重复的因果块；它没有渐近平直时空的 $i^0$ 与 $\mathscr I^\pm$，其 dS 型过去、未来共形边界是类空的。

\subsection{AdS 时空}

在五维平直空间 $\mathbb R^{2,3}$ 中取
\[
    \d s_5^2=-(\d X^{-1})^2-(\d X^0)^2
      +(\d X^1)^2+(\d X^2)^2+(\d X^3)^2.
\]
双曲面
\begin{equation}\label{eq:AdS-embedding}
    -(X^{-1})^2-(X^0)^2+\sum_{i=1}^3(X^i)^2=-L^2
\end{equation}
的诱导度规是四维 \textbf{anti-de Sitter 时空}。取
\[
    X^{-1}=L\cosh\rho\cos\tau,\qquad
    X^0=L\cosh\rho\sin\tau,
    \qquad X^i=L\sinh\rho\,n^i,
\]
其中 $\bm n\in S^2$，可得
\begin{equation}\label{eq:AdS-global-rho}
    \d s^2=L^2\left(-\cosh^2\rho\,\d\tau^2
      +\d\rho^2+\sinh^2\rho\,\d\Omega^2\right).
\end{equation}
嵌入坐标使 $\tau$ 具有周期 $2\pi$，故原双曲面上固定 $\rho,\bm n$ 的曲线是闭合类时线。物理上通常取其\textbf{万有覆盖}(universal cover)，把 $\tau$ 展开为 $\mathbb R$；下文所谓 AdS 均指这个无闭合类时线的覆盖空间。

令 $t=L\tau$、$r=L\sinh\rho$，则全局度规化为
\begin{equation}\label{eq:AdS-global-static}
    \d s^2=-\left(1+\frac{r^2}{L^2}\right)\d t^2
      +\left(1+\frac{r^2}{L^2}\right)^{-1}\d r^2+r^2\d\Omega^2.
\end{equation}
与 dS 静态形式相比，符号的改变使 $\partial_t$ 在所有有限 $r$ 上都为类时，因此纯 AdS 没有宇宙学视界。再令 $r=L\tan\chi$，$0\leqslant\chi<\pi/2$，便有
\begin{equation}\label{eq:AdS-conformal}
    \d s^2=\frac{L^2}{\cos^2\chi}
      \left(-\d\tau^2+\d\chi^2+\sin^2\chi\,\d\Omega^2\right).
\end{equation}
共形边界位于 $\chi=\pi/2$，其共形度规为 $-\d\tau^2+\d\Omega^2$，拓扑是 $\mathbb R\times S^2$。它是\textbf{类时边界}：径向光线满足 $\d\chi=\pm\d\tau$，从中心到边界只需全局坐标时间 $\Delta\tau=\pi/2$，尽管边界在物理度规下仍位于无限空间距离处。

图 \ref{fig:AdS-penrose} 也说明 AdS 不是整体双曲时空。任取一个无边界的类空切片，总能有因果信号从类时共形边界进入而不与该切片相交，所以初值本身不能唯一决定未来；还必须在 $\mathscr I$ 指定反射、透明或其他适当边界条件。AdS 的“盒状”性质正来自信号可抵达边界并在反射条件下返回，而不是来自有限的物理空间体积。

另一套常用坐标是 Poincar\'e 坐标：
\begin{equation}\label{eq:AdS-poincare}
    \d s^2=\frac{L^2}{z^2}
      \left(\d z^2-\d t^2+\d x^2+\d y^2\right),
    \qquad z>0.
\end{equation}
$z=0$ 是共形边界的一部分，其代表度规为三维 Minkowski 度规；$z\to\infty$ 是 Poincar\'e 视界。两者都不是曲率奇性，这套坐标只覆盖全局 AdS 的一个区域。AdS 等距群 $SO(2,3)$ 在边界上恰表现为三维时空的共形群，这是 AdS/CFT 对应的几何起点。

最后，负宇宙学常数也显著改变黑洞热力学。四维 Schwarzschild--AdS 解为
\begin{equation}\label{eq:schwarzschild-AdS}
    f(r)=1-\frac{2M}{r}+\frac{r^2}{L^2},
    \qquad
    T_H=\frac{\hbar}{4\pi r_h}
      \left(1+\frac{3r_h^2}{L^2}\right),
\end{equation}
其中 $r_h$ 是 $f(r_h)=0$ 的唯一正根。$r_h<L/\sqrt3$ 的小黑洞具有负热容，而 $r_h>L/\sqrt3$ 的大黑洞具有正热容，可以与 AdS 边界条件所形成的热库稳定平衡；在正则系综中还会出现热 AdS 与大黑洞之间的 Hawking--Page 相变。由此可见，$\Lambda$ 的符号不仅改变局域曲率，也决定无穷远的类型、视界结构以及允许的热力学系综。
